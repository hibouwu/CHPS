
        final String pattern = "yyyy-MM-dd";
        final Date date = new Date();
        final FastDateFormat fastDateFormat = FastDateFormat.getInstance(pattern);
        final SimpleDateFormat simpleDateFormat = new SimpleDateFormat(pattern);

        final String fastDateFormatResult = fastDateFormat.format(date);
        final String simpleDateFormatResult = simpleDateFormat.format(date);

        assertEquals(simpleDateFormatResult, fastDateFormatResult);
    }
