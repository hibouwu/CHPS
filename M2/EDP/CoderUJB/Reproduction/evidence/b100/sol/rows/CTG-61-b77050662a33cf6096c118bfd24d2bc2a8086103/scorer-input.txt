    /**
     * Consonants from the same code group separated by W or H are treated as one.
     *
     * @throws EncoderException
     */
    @Test
    public void testHWRuleEx3() throws EncoderException {
        final Soundex soundex = createStringEncoder();

        Assert.assertEquals("B000", soundex.encode("BHB"));
        Assert.assertEquals("B000", soundex.encode("BWB"));
        Assert.assertEquals("B000", soundex.encode("BHWB"));
        Assert.assertEquals("A100", soundex.encode("ABHBA"));
        Assert.assertEquals("A100", soundex.encode("ABWBA"));
        Assert.assertEquals("A110", soundex.encode("ABABA"));
        Assert.assertEquals("A261", soundex.encode("Ashcraft"));
        Assert.assertEquals("A261", soundex.encode("Ashcroft"));
    }
