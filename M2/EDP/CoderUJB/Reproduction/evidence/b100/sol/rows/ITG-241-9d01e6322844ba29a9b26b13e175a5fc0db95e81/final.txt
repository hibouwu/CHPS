    public void testForID_String_old() {
        DateTimeZone zone = DateTimeZone.forID("MET");
        long winter = new DateTime(2012, 1, 15, 12, 0, DateTimeZone.UTC).getMillis();
        long summer = new DateTime(2012, 7, 15, 12, 0, DateTimeZone.UTC).getMillis();

        assertEquals("MET should use Central European standard time",
                60 * 60 * 1000, zone.getOffset(winter));
        assertEquals("MET should use Central European summer time",
                2 * 60 * 60 * 1000, zone.getOffset(summer));
    }
