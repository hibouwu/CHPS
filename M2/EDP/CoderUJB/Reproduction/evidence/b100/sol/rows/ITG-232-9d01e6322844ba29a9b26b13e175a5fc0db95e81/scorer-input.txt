    public void testCodec112() {
        byte[] input = new byte[] { 1, 2, 3 };
        byte[] expected = new byte[] { 'A', 'Q', 'I', 'D' };

        byte[] encoded = Base64.encodeBase64(input, false, false, 4);
        assertTrue(java.util.Arrays.equals(expected, encoded));

        encoded = Base64.encodeBase64(input, false, true, 4);
        assertTrue(java.util.Arrays.equals(expected, encoded));

        try {
            Base64.encodeBase64(input, false, false, 3);
            fail("Expected IllegalArgumentException when maxResultSize is too small");
        } catch (IllegalArgumentException expectedException) {
            // Expected: four encoded bytes cannot fit within a limit of three.
        }
    }
