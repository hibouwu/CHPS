@Override
public JsonToken nextToken() throws IOException
{
    // First: any buffered tokens to emit?
    TokenFilterContext ctxt = _exposedContext;
    if (ctxt != null) {
        while (true) {
            JsonToken t = ctxt.nextTokenToRead();
            if (t != null) {
                _currToken = t;
                return t;
            }
            // All buffered tokens for this context have been emitted
            if (ctxt == _headContext) {
                _exposedContext = null;
                if (ctxt.inArray()) {
                    // Return the START_ARRAY token that opened this buffered context
                    _currToken = delegate.getCurrentToken();
                    return _currToken;
                }
                // For objects we simply fall through to read more
                break;
            }
            // Move to the parent context
            ctxt = _headContext.findChildOf(ctxt);
            _exposedContext = ctxt;
            if (ctxt == null) {
                throw _constructError("Unexpected problem: chain of filtered context broken");
            }
        }
    }

    // No buffered tokens, read the next token from the delegate parser
    JsonToken t = delegate.nextToken();
    if (t == null) {
        return (_currToken = null);
    }

    TokenFilter f;

    switch (t.id()) {
    case ID_START_ARRAY:
        f = _itemFilter;
        if (f == TokenFilter.INCLUDE_ALL) {
            _headContext = _headContext.createChildArrayContext(f, true);
            return (_currToken = t);
        }
        if (f == null) {
            delegate.skipChildren();
            break;
        }
        f = _headContext.checkValue(f);
        if (f == null) {
            delegate.skipChildren();
            break;
        }
        if (f != TokenFilter.INCLUDE_ALL) {
            f = f.filterStartArray();
        }
        _itemFilter = f;
        if (f == TokenFilter.INCLUDE_ALL) {
            _headContext = _headContext.createChildArrayContext(f, true);
            return (_currToken = t);
        }
        _headContext = _headContext.createChildArrayContext(f, false);
        if (_includePath) {
            JsonToken buffered = _nextTokenWithBuffering(_headContext);
            if (buffered != null) {
                _currToken = buffered;
                return buffered;
            }
        }
        break;

    case ID_START_OBJECT:
        f = _itemFilter;
        if (f == TokenFilter.INCLUDE_ALL) {
            _headContext = _headContext.createChildObjectContext(f, true);
            return (_currToken = t);
        }
        if (f == null) {
            delegate.skipChildren();
            break;
        }
        f = _headContext.checkValue(f);
        if (f == null) {
            delegate.skipChildren();
            break;
        }
        if (f != TokenFilter.INCLUDE_ALL) {
            f = f.filterStartObject();
        }
        _itemFilter = f;
        if (f == TokenFilter.INCLUDE_ALL) {
            _headContext = _headContext.createChildObjectContext(f, true);
            return (_currToken = t);
        }
        _headContext = _headContext.createChildObjectContext(f, false);
        if (_includePath) {
            JsonToken buffered = _nextTokenWithBuffering(_headContext);
            if (buffered != null) {
                _currToken = buffered;
                return buffered;
            }
        }
        break;

    case ID_END_ARRAY:
    case ID_END_OBJECT:
        {
            boolean returnEnd = _headContext.isStartHandled();
            f = _headContext.getFilter();
            if (f != null && f != TokenFilter.INCLUDE_ALL) {
                if (t.id() == ID_END_ARRAY) {
                    f.filterFinishArray();
                } else {
                    f.filterFinishObject();
                }
            }
            _headContext = _headContext.getParent();
            _itemFilter = _headContext.getFilter();
            if (returnEnd) {
                return (_currToken = t);
            }
        }
        break;

    case ID_FIELD_NAME:
        {
            final String name = delegate.getCurrentName();
            f = _headContext.setFieldName(name);
            if (f == TokenFilter.INCLUDE_ALL) {
                _itemFilter = f;
                if (!_includePath) {
                    if (_includeImmediateParent && !_headContext.isStartHandled()) {
                        JsonToken start = _headContext.nextTokenToRead(); // START_OBJECT
                        _exposedContext = _headContext;
                        _currToken = start;
                        return start;
                    }
                }
                return (_currToken = t);
            }
            if (f == null) {
                delegate.nextToken(); // move to value
                delegate.skipChildren();
                break;
            }
            f = f.includeProperty(name);
            if (f == null) {
                delegate.nextToken(); // move to value
                delegate.skipChildren();
                break;
            }
            _itemFilter = f;
            if (f == TokenFilter.INCLUDE_ALL && _includePath) {
                return (_currToken = t);
            }
            if (_includePath) {
                JsonToken buffered = _nextTokenWithBuffering(_headContext);
                if (buffered != null) {
                    _currToken = buffered;
                    return buffered;
                }
            }
        }
        break;

    default: // scalar values
        f = _itemFilter;
        if (f == TokenFilter.INCLUDE_ALL) {
            return (_currToken = t);
        }
        if (f != null) {
            f = _headContext.checkValue(f);
            if (f == TokenFilter.INCLUDE_ALL || (f != null && f.includeValue(delegate))) {
                return (_currToken = t);
            }
        }
        // not included
        break;
    }

    // If we reach here, the current token was not emitted; try again
    return _nextToken2();
}
