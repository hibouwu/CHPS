    public void testFormat() {
        Calendar cal = Calendar.getInstance();
        cal.set(2003, Calendar.JANUARY, 1, 0, 0, 0);
        cal.set(Calendar.MILLISECOND, 0);
        Date date = cal.getTime();

        SimpleDateFormat sdf = new SimpleDateFormat("Y");
        String expected = sdf.format(date);
        FastDateFormat fdf = FastDateFormat.getInstance("Y");
        String actual = fdf.format(date);
        assertEquals("FastDateFormat should format year 'Y' same as SimpleDateFormat", expected, actual);

        sdf.applyPattern("YYY");
        expected = sdf.format(date);
        fdf = FastDateFormat.getInstance("YYY");
        actual = fdf.format(date);
        assertEquals("FastDateFormat should format year 'YYY' same as SimpleDateFormat", expected, actual);
    }
