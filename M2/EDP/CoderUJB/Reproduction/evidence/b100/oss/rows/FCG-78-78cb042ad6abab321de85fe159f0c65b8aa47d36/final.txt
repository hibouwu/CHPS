    private static NodePointer createChildPointerForStep(
                PropertyOwnerPointer parentPointer, Step step) {
        NodeTest test = step.getNodeTest();

        // Attribute test (e.g., @lang)
        if (test instanceof NameAttributeTest) {
            QName name = ((NameAttributeTest) test).getName();
            if (isLangAttribute(name)) {
                // Special handling for xml:lang attribute
                return new LangAttributePointer(parentPointer);
            }
            // General attribute handling
            String attrName = name.getName();
            NodePointer attrPointer = parentPointer.getAttributePointer(attrName);
            if (attrPointer == null) {
                // If the attribute does not exist, return a null pointer
                attrPointer = new NullPropertyPointer(parentPointer, attrName);
            }
            return attrPointer;
        }

        // Element/property name test
        if (test instanceof NodeNameTest) {
            QName name = ((NodeNameTest) test).getNodeName();
            String propName = name.getName();
            NodePointer child = parentPointer.getPropertyPointer(propName);
            if (child == null) {
                // Property does not exist – create a null element pointer
                child = new NullPropertyPointer(parentPointer, propName);
            }
            return child;
        }

        // Fallback: let the parent pointer create a generic child pointer
        // (covers cases like wildcard or node type tests)
        try {
            // Most PropertyOwnerPointer implementations provide a
            // createNodePointer(Step) method.
            java.lang.reflect.Method m = parentPointer.getClass()
                    .getMethod("createNodePointer", Step.class);
            return (NodePointer) m.invoke(parentPointer, step);
        } catch (Exception e) {
            // As a last resort, return a null element pointer
            return new NullElementPointer(parentPointer);
        }
    }
