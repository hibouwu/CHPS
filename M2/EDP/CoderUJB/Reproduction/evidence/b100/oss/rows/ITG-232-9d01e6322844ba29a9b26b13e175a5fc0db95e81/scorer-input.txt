public void testCodec112() {
    byte[] binaryData = "sample data".getBytes(java.nio.charset.StandardCharsets.UTF_8);
    int maxResultSize = 1024; // a valid size larger than the expected output
    org.junit.jupiter.api.Assertions.assertDoesNotThrow(() -> {
        byte[] encoded = Base64.encodeBase64(binaryData, false, false, maxResultSize);
        org.junit.jupiter.api.Assertions.assertNotNull(encoded);
        // Optionally verify that the result length does not exceed maxResultSize
        org.junit.jupiter.api.Assertions.assertTrue(encoded.length <= maxResultSize);
    });
}
