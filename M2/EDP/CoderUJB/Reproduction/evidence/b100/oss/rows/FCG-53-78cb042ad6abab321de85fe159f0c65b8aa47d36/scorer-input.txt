    private Node transformAsString(AstNode n) {
        // First transform the node using the generic transformation logic.
        Node result = justTransform(n);
        int type = result.getType();

        if (type == Token.NAME) {
            // Convert a Name token to a STRING token (unquoted).
            // The identifier text is stored in the node's string field.
            result = newStringNode(result.getString());
        } else if (type == Token.STRING) {
            // The node is already a STRING token; ensure it is quoted.
            // Create a new STRING node with the same value and mark it as quoted.
            Node quoted = newStringNode(result.getString());
            quoted.setQuotedString(true);
            result = quoted;
        }
        return result;
    }
