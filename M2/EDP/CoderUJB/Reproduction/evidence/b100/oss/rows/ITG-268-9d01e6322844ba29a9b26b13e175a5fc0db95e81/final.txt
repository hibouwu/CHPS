@Test
public void supportsBOMinFiles() throws IOException {
    // UTF-16LE BOM bytes
    byte[] bom = new byte[] {(byte)0xFF, (byte)0xFE};
    String expected = "Hello – world";
    byte[] content = expected.getBytes(StandardCharsets.UTF_16LE);
    byte[] data = new byte[bom.length + content.length];
    System.arraycopy(bom, 0, data, 0, bom.length);
    System.arraycopy(content, 0, data, bom.length, content.length);
    try (InputStream in = new ByteArrayInputStream(data)) {
        Document doc = Jsoup.parse(in, null, "http://example.com/");
        assertEquals(expected, doc.body().text());
    }
}
