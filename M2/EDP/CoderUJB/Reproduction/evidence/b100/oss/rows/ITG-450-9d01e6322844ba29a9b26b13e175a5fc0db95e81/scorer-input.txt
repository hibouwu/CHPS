    @Test
    public void shouldMatchCaptureArgumentsWhenArgsCountDoesNOTMatch() throws Exception {
        // Locate the Mockito core JAR
        java.net.URL location = org.mockito.Mockito.class.getProtectionDomain().getCodeSource().getLocation();
        java.io.File jarFile = new java.io.File(location.toURI());
        org.junit.Assert.assertTrue("Mockito JAR should exist as a file", jarFile.isFile());

        try (java.util.jar.JarFile jar = new java.util.jar.JarFile(jarFile)) {
            java.util.Enumeration<? extends java.util.jar.JarEntry> entries = jar.entries();
            while (entries.hasMoreElements()) {
                java.util.jar.JarEntry entry = entries.nextElement();
                org.junit.Assert.assertFalse(
                    "Source file should not be packaged in the binary JAR: " + entry.getName(),
                    entry.getName().endsWith(".java"));
            }
        }
    }
