    /**
     * Test case for
     * <a href="https://issues.apache.org/jira/browse/COMPRESS-93"
     * >COMPRESS-93</a>.
     */
    @Test
    public void testCompressionMethod() throws Exception {
        ZipArchiveEntry entry = new ZipArchiveEntry("test.txt");
        
        // Default compression method should be DEFLATED
        assertEquals(ZipMethod.DEFLATED_CODE, entry.getMethod());
        
        // Set compression method to STORED
        entry.setMethod(ZipMethod.STORED_CODE);
        assertEquals(ZipMethod.STORED_CODE, entry.getMethod());
        
        // Set compression method back to DEFLATED
        entry.setMethod(ZipMethod.DEFLATED_CODE);
        assertEquals(ZipMethod.DEFLATED_CODE, entry.getMethod());
        
        // Test with an invalid method code
        try {
            entry.setMethod(-1);
            fail("Expected IllegalArgumentException for invalid method code");
        } catch (IllegalArgumentException e) {
            // Expected exception
        }
    }
