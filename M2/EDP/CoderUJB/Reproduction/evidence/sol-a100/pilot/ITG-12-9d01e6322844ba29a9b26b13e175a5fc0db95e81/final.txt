public void testEscapeHtmlHighUnicode() throws java.io.UnsupportedEncodingException {
    // U+1D362: COUNTING ROD UNIT DIGIT THREE, encoded as UTF-8.
    byte[] data = new byte[] {
        (byte) 0xF0, (byte) 0x9D, (byte) 0x8D, (byte) 0xA2
    };
    String input = new String(data, "UTF-8");

    assertEquals("Supplementary characters must be escaped as a single code point",
            "&#119650;", StringEscapeUtils.escapeHtml(input));
}
