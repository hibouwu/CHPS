/**
 * Test unequal comparisons with second comparators.
 */
public void testSecond() {
    aDateTime = new DateTime(2004, 6, 9, 12, 30, 15, 250, DateTimeZone.UTC);
    bDateTime = aDateTime.plusSeconds(1);

    assertEquals("SecondM1", -1, cSecond.compare(aDateTime, bDateTime));
    assertEquals("SecondP1", 1, cSecond.compare(bDateTime, aDateTime));

    // Millisecond differences within the same second are ignored.
    bDateTime = aDateTime.plusMillis(500);
    assertEquals("SecondEqual", 0, cSecond.compare(aDateTime, bDateTime));
}
