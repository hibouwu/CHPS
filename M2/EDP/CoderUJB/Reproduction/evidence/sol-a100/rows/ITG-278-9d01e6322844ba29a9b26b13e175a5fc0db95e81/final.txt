import org.junit.Test;

import java.util.List;

import static org.mockito.Mockito.*;

public class Mockito138Test {

    @Test
    @SuppressWarnings("unchecked")
    public void shouldAllowVerifyingWhenOtherMockCallIsInTheSameLine() {
        List<String> mock = mock(List.class);
        List<String> otherMock = mock(List.class);
        when(otherMock.get(0)).thenReturn("value");

        mock.add("value");

        // The intervening call must not consume verification intended for mock.
        verify(mock).add(otherMock.get(0));
    }
}
