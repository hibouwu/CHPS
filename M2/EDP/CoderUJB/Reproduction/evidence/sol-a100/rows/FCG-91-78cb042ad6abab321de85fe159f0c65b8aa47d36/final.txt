public String soundex(String str) {
    if (str == null) {
        return null;
    }

    str = SoundexUtils.clean(str);
    if (str.isEmpty()) {
        return str;
    }

    final char[] code = { '0', '0', '0', '0' };
    code[0] = str.charAt(0);

    char lastCode = map(code[0]);
    int count = 1;

    for (int i = 1; i < str.length() && count < code.length; i++) {
        final char mapped = map(str.charAt(i));

        // Silent letters do not separate adjacent matching codes.
        if (mapped == '#') {
            continue;
        }

        if (mapped != '0' && mapped != lastCode) {
            code[count++] = mapped;
        }

        lastCode = mapped;
    }

    return new String(code);
}
