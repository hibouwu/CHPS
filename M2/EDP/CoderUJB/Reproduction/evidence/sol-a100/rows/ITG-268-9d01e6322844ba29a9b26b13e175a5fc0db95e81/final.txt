import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.junit.Test;

import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;

import static org.junit.Assert.assertEquals;

public class DataUtilTest {
    @Test
    public void supportsBOMinFiles() throws IOException {
        String[] encodings = {"UTF-8", "UTF-16LE", "UTF-16BE"};
        byte[][] boms = {
            {(byte) 0xEF, (byte) 0xBB, (byte) 0xBF},
            {(byte) 0xFF, (byte) 0xFE},
            {(byte) 0xFE, (byte) 0xFF}
        };

        String title = "BOM encoding test";
        String text = "H\u00e9llo \u4e16\u754c";
        String html = "<html><head><title>" + title + "</title></head>"
                + "<body><p>" + text + "</p></body></html>";

        for (int i = 0; i < encodings.length; i++) {
            File file = File.createTempFile("jsoup-bom-", ".html");
            try {
                try (FileOutputStream out = new FileOutputStream(file)) {
                    out.write(boms[i]);
                    out.write(html.getBytes(encodings[i]));
                }

                // No explicit charset: the parser must detect the raw BOM.
                Document doc = Jsoup.parse(file, null, "");

                assertEquals(encodings[i], title, doc.title());
                assertEquals(encodings[i], text, doc.body().text());
                // The BOM must not appear as document text.
                assertEquals(encodings[i], title + " " + text, doc.text());
            } finally {
                if (!file.delete()) {
                    file.deleteOnExit();
                }
            }
        }
    }
}
