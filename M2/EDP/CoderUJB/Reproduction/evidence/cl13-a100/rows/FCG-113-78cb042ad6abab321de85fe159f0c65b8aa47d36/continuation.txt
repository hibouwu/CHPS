    /**

    public static byte[] mergeCentralDirectoryData(final ZipExtraField[] data) {
        final List<byte[]> result = new ArrayList<>();
        for (final ZipExtraField field : data) {
            final byte[] centralDirectory = field.getCentralDirectoryData();
            if (centralDirectory != null) {
                result.add(centralDirectory);
            }
        }
        return mergeArrays(result);
    }
