
        if (str == null) {
            return null;
        }
        str = SoundexUtils.clean(str);
        if (str.length() == 0) {
            return str;
        }
        char[] in = str.toUpperCase().toCharArray();
        char[] out = new char[4];
        out[0] = in[0];
        int length = 1;
        for (int i = 1; i < in.length && length < out.length; i++) {
            char c = map(in[i]);
            if (c != 0 && c != out[length - 1]) {
                out[length] = c;
                length++;
            }
        }
        return new String(out, 0, length);
    }
