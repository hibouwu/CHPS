            throw new IllegalArgumentException("count must be between 1 and 63");

        if (count > bitsCachedSize) {
            if (bitsCachedSize == 0) {
                bitsCached = in.read();
                if (bitsCached == -1) {
                    return -1;
                }
                bitsCachedSize = 8;
            }
            if (count > bitsCachedSize) {
                bitsCached = (bitsCached << 8) | in.read();
                if (bitsCached == -1) {
                    return -1;
                }
                bitsCachedSize += 8;
            }
        }
