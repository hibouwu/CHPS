private Token parseSimpleToken(final Token token, int ch) throws IOException {
    final StringBuilder buffer = new StringBuilder();

    while (ch != END_OF_STREAM) {
        if (isEndOfFile(ch)) {
            // End of file reached
            token.type = EOF;
            break;
        } else if (isEndOfLine(ch)) {
            // End of line reached
            token.type = EORECORD;
            if (firstEol == null) {
                firstEol = String.valueOf((char) ch);
                if (ch == CR && reader.peek() == LF) {
                    // CRLF
                    firstEol += (char) reader.read();
                }
            }
            break;
        } else if (isDelimiter(ch)) {
            // Unescaped delimiter reached
            token.type = TOKEN;
            break;
        } else if (isEscape(ch)) {
            // Handle escape character
            ch = readEscape();
            if (ch == END_OF_STREAM) {
                token.type = EOF;
                break;
            }
            buffer.append((char) ch);
        } else {
            // Regular character
            buffer.append((char) ch);
        }

        ch = reader.read();
    }

    token.content = buffer.toString();
    return token;
}
