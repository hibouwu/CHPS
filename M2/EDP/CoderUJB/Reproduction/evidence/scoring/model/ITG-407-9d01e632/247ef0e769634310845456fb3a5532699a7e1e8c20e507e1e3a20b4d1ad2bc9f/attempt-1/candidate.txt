public void testUnexpectedToken() throws Exception {
        Gson gson = new GsonBuilder().setDateFormat("yyyy-MM-dd'T'HH:mm:ssZ").create();

        // Test serialization of null date
        Date nullDate = null;
        String serializedNullDate = gson.toJson(nullDate);
        assertEquals("null", serializedNullDate);

        // Test deserialization of null date
        Date deserializedNullDate = gson.fromJson("null", Date.class);
        assertNull(deserializedNullDate);

        // Test serialization of a valid date
        Date validDate = new Date();
        String serializedValidDate = gson.toJson(validDate);
        assertNotNull(serializedValidDate);

        // Test deserialization of a valid date
        Date deserializedValidDate = gson.fromJson(serializedValidDate, Date.class);
        assertNotNull(deserializedValidDate);
        assertEquals(validDate.getTime(), deserializedValidDate.getTime());

        // Test deserialization of an invalid date string (should throw JsonSyntaxException)
        try {
            gson.fromJson("\"invalid-date\"", Date.class);
            fail("Expected JsonSyntaxException for invalid date string");
        } catch (JsonSyntaxException e) {
            // Expected exception
        }

        // Test deserialization of a non-string token (should throw JsonSyntaxException)
        try {
            gson.fromJson("12345", Date.class);
            fail("Expected JsonSyntaxException for non-string token");
        } catch (JsonSyntaxException e) {
            // Expected exception
        }
    }
