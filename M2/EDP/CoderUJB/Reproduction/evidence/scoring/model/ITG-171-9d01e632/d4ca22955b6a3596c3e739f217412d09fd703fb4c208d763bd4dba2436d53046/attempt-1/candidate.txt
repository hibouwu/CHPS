@Test
public void testHWRuleEx1() {
        // Test case to verify the HW rule in Soundex
        // According to the rule, if two characters that map to the same code are separated by H or W,
        // the second one should be encoded.
        
        // Example: "Ashcraft" should be encoded as "A261" not "A216"
        // 'S' and 'C' both map to '2', and they are separated by 'H' and 'R', so 'C' should be encoded.
        assertEquals("A261", soundex.encode("Ashcraft"));
        
        // Another example: "Gough" should be encoded as "G200" not "G250"
        // 'O' and 'U' both map to '0', and they are separated by 'G' and 'H', so 'U' should be encoded.
        assertEquals("G200", soundex.encode("Gough"));
        
        // Example: "Hewitt" should be encoded as "H530" not "H503"
        // 'W' separates 'H' and 'I', and 'I' and 'T', but 'I' and 'T' map to different codes, so both should be encoded.
        assertEquals("H530", soundex.encode("Hewitt"));
        
        // Example: "Hill" should be encoded as "H400" not "H400"
        // 'L' and 'L' map to '4', and they are separated by 'I', so the second 'L' should be encoded.
        assertEquals("H400", soundex.encode("Hill"));
        
        // Example: "Wright" should be encoded as "W620" not "W602"
        // 'R' and 'G' both map to '6', and they are separated by 'I' and 'G', so 'G' should be encoded.
        assertEquals("W620", soundex.encode("Wright"));
    }
