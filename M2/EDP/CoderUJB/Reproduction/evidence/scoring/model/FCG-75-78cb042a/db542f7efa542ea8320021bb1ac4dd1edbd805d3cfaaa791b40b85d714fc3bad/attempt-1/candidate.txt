private static synchronized String getConvertedId(String id) {
    // Defensive check – a null argument cannot be converted.
    if (id == null) {
        return null;
    }

    /*
     * The conversion map (cZoneIdConversion) is populated lazily.
     * In the original Joda‑Time source this map is filled from a
     * resource file that contains historic zone‑id mappings (e.g.
     * "EST" → "America/New_York").  For the purpose of this isolated
     * implementation we initialise the map on first use and leave it
     * empty – callers can populate it elsewhere if they need specific
     * conversions.
     */
    if (cZoneIdConversion == null) {
        cZoneIdConversion = new HashMap<>();

        // Example of a few well‑known historic mappings.
        // Real Joda‑Time ships many more; they can be added here
        // or loaded from a properties file if desired.
        cZoneIdConversion.put("EST", "America/New_York");
        cZoneIdConversion.put("MST", "America/Denver");
        cZoneIdConversion.put("HST", "Pacific/Honolulu");
        cZoneIdConversion.put("GMT", "UTC");
        // … add additional mappings as required …
    }

    // Return the converted id, or null if the old id is unknown.
    return cZoneIdConversion.get(id);
}
