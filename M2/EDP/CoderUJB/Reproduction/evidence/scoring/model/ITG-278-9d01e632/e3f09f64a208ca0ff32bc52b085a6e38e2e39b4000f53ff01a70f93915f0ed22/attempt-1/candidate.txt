@Test
public void shouldAllowVerifyingWhenOtherMockCallIsInTheSameLine() {
        // -----------------------------------------------------------------
        // Arrange – create two independent mocks
        // -----------------------------------------------------------------
        List<String> firstMock  = mock(List.class);
        List<String> secondMock = mock(List.class);

        // -----------------------------------------------------------------
        // Act – invoke a method on the first mock
        // -----------------------------------------------------------------
        firstMock.add("one");

        // -----------------------------------------------------------------
        // Assert – verification and another mock call are placed on the SAME line.
        // The verification must succeed even though a second mock call follows it
        // on the same source line (separated by a semicolon).
        // -----------------------------------------------------------------
        verify(firstMock).add("one"); secondMock.clear();

        // Additionally verify that the second mock call really happened.
        verify(secondMock).clear();
    }
