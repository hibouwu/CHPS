public void testMapToProperties() throws Exception {
    ObjectMapper mapper = new ObjectMapper();
    Map<String, Object> source = new HashMap<>();
    source.put("int", 123);
    source.put("bool", true);
    source.put("string", "value");
    String json = mapper.writeValueAsString(source);
    Properties props = mapper.readValue(json, Properties.class);
    assertEquals("123", props.getProperty("int"));
    assertEquals("true", props.getProperty("bool"));
    assertEquals("value", props.getProperty("string"));
    for (String name : props.stringPropertyNames()) {
        assertTrue(props.get(name) instanceof String);
    }
}
