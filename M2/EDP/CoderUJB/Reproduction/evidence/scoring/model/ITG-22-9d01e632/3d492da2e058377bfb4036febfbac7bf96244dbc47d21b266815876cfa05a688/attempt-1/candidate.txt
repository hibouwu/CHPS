public void testMapToProperties() throws Exception {
    ObjectMapper mapper = new ObjectMapper();

    Map<String, Object> input = new LinkedHashMap<String, Object>();
    input.put("text", "value");
    input.put("integer", Integer.valueOf(42));
    input.put("decimal", Double.valueOf(1.25));
    input.put("boolean", Boolean.TRUE);

    Properties properties = mapper.convertValue(input, Properties.class);

    assertEquals(input.size(), properties.size());
    assertEquals("value", properties.getProperty("text"));
    assertEquals("42", properties.getProperty("integer"));
    assertEquals("1.25", properties.getProperty("decimal"));
    assertEquals("true", properties.getProperty("boolean"));

    // Properties must contain actual Strings, not the original value types.
    for (Object value : properties.values()) {
        assertTrue(value instanceof String);
    }
}
