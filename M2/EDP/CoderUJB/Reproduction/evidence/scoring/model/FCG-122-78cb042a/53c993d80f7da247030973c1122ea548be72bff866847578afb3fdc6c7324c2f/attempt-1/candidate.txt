  public Symbol getParameterInFunction(Symbol sym, String paramName) {
    if (sym.getType() instanceof FunctionType) {
      FunctionType fn = sym.getType().toMaybeFunctionType();
      for (Symbol param : fn.getParameters()) {
        if (param.getName().equals(paramName)) {
          return param;
        }
      }
    }
    return null;
  }
