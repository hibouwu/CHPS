@Test
public void testRfc4648Section10Decode() {
    // RFC‑4648 Section 10 test vectors: {plain‑text, base64‑encoded}
    final String[][] vectors = {
        {"",        ""},
        {"f",       "Zg=="},
        {"fo",      "Zm8="},
        {"foo",     "Zm9v"},
        {"foob",    "Zm9vYg=="},
        {"fooba",   "Zm9vYmE="},
        {"foobar",  "Zm9vYmFy"}
    };

    for (final String[] pair : vectors) {
        final String plain   = pair[0];
        final String encoded = pair[1];

        // Decode the Base64 string
        final byte[] decoded = Base64.decodeBase64(encoded);

        // Basic sanity checks
        assertNotNull("Decoded byte array must not be null for '" + encoded + "'", decoded);
        assertEquals("Decoded length mismatch for '" + encoded + "'",
                     plain.getBytes(CHARSET_UTF8).length,
                     decoded.length);

        // Verify that the decoded bytes represent the original plain‑text
        final String roundTrip = new String(decoded, CHARSET_UTF8);
        assertEquals("Decoded value does not match original for '" + encoded + "'",
                     plain,
                     roundTrip);
    }
}
