    @Test
    public void testRfc4648Section10Decode() {
        assertEquals("", new String(Base64.decodeBase64(""), CHARSET_UTF8));
        assertEquals("f", new String(Base64.decodeBase64("Zg=="), CHARSET_UTF8));
        assertEquals("fo", new String(Base64.decodeBase64("Zm8="), CHARSET_UTF8));
        assertEquals("foo", new String(Base64.decodeBase64("Zm9v"), CHARSET_UTF8));
        assertEquals("foob", new String(Base64.decodeBase64("Zm9vYg=="), CHARSET_UTF8));
        assertEquals("fooba", new String(Base64.decodeBase64("Zm9vYmE="), CHARSET_UTF8));
        assertEquals("foobar", new String(Base64.decodeBase64("Zm9vYmFy"), CHARSET_UTF8));
    }
