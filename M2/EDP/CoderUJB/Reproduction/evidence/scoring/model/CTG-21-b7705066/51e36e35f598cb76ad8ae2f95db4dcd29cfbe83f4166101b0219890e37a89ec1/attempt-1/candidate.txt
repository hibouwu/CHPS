    @Test
    public void testRfc4648Section10Decode() {
        assertEquals("", Base64.decodeBase64(""));
        assertEquals("f", Base64.decodeBase64("Zg=="));
        assertEquals("fo", Base64.decodeBase64("Zm8="));
        assertEquals("foo", Base64.decodeBase64("Zm9v"));
        assertEquals("foob", Base64.decodeBase64("Zm9vYg=="));
        assertEquals("fooba", Base64.decodeBase64("Zm9vYmE="));
        assertEquals("foobar", Base64.decodeBase64("Zm9vYmFy"));
    }
