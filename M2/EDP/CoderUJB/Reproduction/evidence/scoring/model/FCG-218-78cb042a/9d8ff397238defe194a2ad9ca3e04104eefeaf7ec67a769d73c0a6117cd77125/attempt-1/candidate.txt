protected JsonGenerator _createUTF8Generator(OutputStream out,
                                            IOContext ctxt) throws IOException
{
    // 1. Apply an {@link OutputDecorator} if one has been configured.
    //    The decorator may replace the stream (e.g. by adding compression)
    //    or wrap it (e.g. by buffering).  The contract of OutputDecorator
    //    is that it returns a non‑null OutputStream.
    if (_outputDecorator != null) {
        out = _outputDecorator.decorate(out, ctxt);
        if (out == null) {
            // Defensive programming – a broken decorator should not hide the
            // problem from the caller.
            throw new IllegalStateException("OutputDecorator returned null OutputStream");
        }
    }

    // 2. Construct the concrete UTF‑8 generator.  In the reference
    //    implementation this is {@code UTF8JsonGenerator}.  It receives
    //    the IOContext, the feature mask and the (possibly decorated) stream.
    JsonGenerator gen = new UTF8JsonGenerator(ctxt,
                                              _generatorFeatures,
                                              _generatorFeatureMask,
                                              out);

    // 3. If the factory has a custom {@link CharacterEscapes} instance,
    //    configure the generator with it.
    if (_characterEscapes != null) {
        gen.setCharacterEscapes(_characterEscapes);
    }

    // 4. If a root‑value separator has been defined, pass it to the generator.
    //    The separator is stored as a {@link SerializableString} (Jackson’s
    //    internal representation of a pre‑encoded string).  If the field is
    //    null we simply leave the generator with its default (no separator).
    if (_rootValueSeparator != null) {
        gen.setRootValueSeparator(_rootValueSeparator);
    }

    // 5. Return the fully‑configured generator.
    return gen;
}
