protected JsonGenerator _createUTF8Generator(OutputStream out, IOContext ctxt) throws IOException {
    // Create a UTF-8 encoded writer from the output stream
    Writer writer = new OutputStreamWriter(out, StandardCharsets.UTF_8);

    // Decorate the writer if necessary
    writer = _decorate(writer, ctxt);

    // Create and return a JsonGenerator instance using the decorated writer
    return new JsonGeneratorImpl(ctxt, writer, this._getBufferRecycler(), this.getGeneratorFeatures());
}
