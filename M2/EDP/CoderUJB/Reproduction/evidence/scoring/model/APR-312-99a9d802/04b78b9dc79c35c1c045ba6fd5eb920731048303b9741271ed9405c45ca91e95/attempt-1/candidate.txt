public static long parseOctal(final byte[] buffer, final int offset,
        final int length) {
    if (length < 2) {
        throw new IllegalArgumentException(
                "Length " + length + " must be at least 2");
    }

    long result = 0L;
    int start = offset;
    int end = offset + length;

    // Missing fields consisting entirely of NULs are valid.
    boolean allNuls = true;
    for (int i = start; i < end; i++) {
        if (buffer[i] != 0) {
            allNuls = false;
            break;
        }
    }
    if (allNuls) {
        return 0L;
    }

    // Require one trailing NUL or space.
    byte currentByte = buffer[end - 1];
    if (currentByte != 0 && currentByte != (byte) ' ') {
        throw new IllegalArgumentException(
                exceptionMessage(buffer, offset, length, end - 1, currentByte));
    }
    end--;

    // Allow one additional trailing NUL or space.
    if (end > start) {
        currentByte = buffer[end - 1];
        if (currentByte == 0 || currentByte == (byte) ' ') {
            end--;
        }
    }

    // Skip leading spaces, but validate every remaining byte.
    while (start < end && buffer[start] == (byte) ' ') {
        start++;
    }

    for (int i = start; i < end; i++) {
        currentByte = buffer[i];
        if (currentByte < '0' || currentByte > '7') {
            throw new IllegalArgumentException(
                    exceptionMessage(buffer, offset, length, i, currentByte));
        }
        result = (result << 3) + (currentByte - '0');
    }

    return result;
}
