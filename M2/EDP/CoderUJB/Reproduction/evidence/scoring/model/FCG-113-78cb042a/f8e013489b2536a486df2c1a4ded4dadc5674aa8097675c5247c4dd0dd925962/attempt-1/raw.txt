public static byte[] mergeLocalFileDataData(final ZipExtraField[] data) {
    final boolean lastIsUnparseable = data.length > 0
            && data[data.length - 1] instanceof UnparseableExtraFieldData;
    final int regularFieldCount = lastIsUnparseable
            ? data.length - 1 : data.length;

    int totalLength = WORD * regularFieldCount;
    for (final ZipExtraField field : data) {
        totalLength += field.getLocalFileDataLength().getValue();
    }

    final byte[] result = new byte[totalLength];
    int offset = 0;

    for (int i = 0; i < regularFieldCount; i++) {
        final ZipExtraField field = data[i];

        System.arraycopy(field.getHeaderId().getBytes(), 0, result, offset, 2);
        System.arraycopy(field.getLocalFileDataLength().getBytes(),
                0, result, offset + 2, 2);
        offset += WORD;

        final byte[] localData = field.getLocalFileDataData();
        if (localData != null) {
            System.arraycopy(localData, 0, result, offset, localData.length);
            offset += localData.length;
        }
    }

    // Unparseable trailing data has no header or length prefix.
    if (lastIsUnparseable) {
        final byte[] localData = data[data.length - 1].getLocalFileDataData();
        if (localData != null) {
            System.arraycopy(localData, 0, result, offset, localData.length);
        }
    }

    return result;
}
