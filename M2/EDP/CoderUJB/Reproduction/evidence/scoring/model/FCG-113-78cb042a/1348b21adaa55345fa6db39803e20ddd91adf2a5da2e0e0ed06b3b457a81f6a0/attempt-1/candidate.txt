public static byte[] mergeLocalFileDataData(final ZipExtraField[] data) {
    if (data == null || data.length == 0) {
        return new byte[0];
    }
    int total = 0;
    for (ZipExtraField field : data) {
        if (field != null) {
            total += field.getLocalFileDataLength().getValue();
        }
    }
    byte[] merged = new byte[total];
    int offset = 0;
    for (ZipExtraField field : data) {
        if (field != null) {
            byte[] bytes = field.getLocalFileDataData();
            int len = field.getLocalFileDataLength().getValue();
            System.arraycopy(bytes, 0, merged, offset, len);
            offset += len;
        }
    }
    return merged;
}
