    public static byte[] mergeLocalFileDataData(final ZipExtraField[] data) {
        final List<byte[]> result = new ArrayList<>();
        for (final ZipExtraField field : data) {
            final byte[] local = field.getLocalFileDataData();
            if (local != null) {
                result.add(local);
            }
        }
        return mergeArrays(result);
    }
