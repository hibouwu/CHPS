public static byte[] mergeLocalFileDataData(final ZipExtraField[] data) {
        List<byte[]> localFileDataList = new ArrayList<>();
        
        for (ZipExtraField field : data) {
            byte[] localFileData = field.getLocalFileDataData();
            if (localFileData != null && localFileData.length > 0) {
                localFileDataList.add(localFileData);
            }
        }

        // Calculate the total length of the merged array
        int totalLength = 0;
        for (byte[] bytes : localFileDataList) {
            totalLength += bytes.length;
        }

        // Create the result array and copy all data into it
        byte[] result = new byte[totalLength];
        int currentIndex = 0;
        for (byte[] bytes : localFileDataList) {
            System.arraycopy(bytes, 0, result, currentIndex, bytes.length);
            currentIndex += bytes.length;
        }

        return result;
    }
