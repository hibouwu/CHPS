public static byte[] mergeLocalFileDataData(final ZipExtraField[] data) {
        // Guard against {@code null} or empty input – the ZIP spec expects an
        // empty extra‑field section in that case.
        if (data == null || data.length == 0) {
            return new byte[0];
        }

        // Use a {@link ByteArrayOutputStream} for efficient concatenation.
        final ByteArrayOutputStream out = new ByteArrayOutputStream();

        for (final ZipExtraField field : data) {
            // Skip any {@code null} entries – they simply contribute nothing.
            if (field == null) {
                continue;
            }

            // The raw data for the *local* header.
            final byte[] localData = field.getLocalFileDataData();

            // Header ID (2 bytes, little‑endian)
            final ZipShort headerId = field.getHeaderId();
            out.write(headerId.getBytes(), 0, 2);

            // Length of the data (2 bytes, little‑endian)
            final ZipShort dataLength = new ZipShort(localData.length);
            out.write(dataLength.getBytes(), 0, 2);

            // The actual data bytes (may be zero‑length)
            out.write(localData, 0, localData.length);
        }

        // Return the accumulated byte array.
        return out.toByteArray();
    }
