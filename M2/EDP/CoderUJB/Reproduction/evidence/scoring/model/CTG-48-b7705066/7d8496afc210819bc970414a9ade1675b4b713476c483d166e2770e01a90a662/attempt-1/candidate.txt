public void testCanInlineReferenceToFunctionInExpression18() {
    // --------------------------------------------------------------------
    // JavaScript source that will be fed to the FunctionInjector.
    // --------------------------------------------------------------------
    //  * foo – pure (annotated with @nosideeffects)
    //  * The call site is: c.test = foo();
    //  * The rest of the script only performs simple object creation /
    //    property assignments that are irrelevant for the purity of foo.
    // --------------------------------------------------------------------
    String js =
        "/** @nosideeffects */ function foo(){return 'foo'};" +
        "c.test = foo();" +                     // <-- the reference we test
        "c = a;" +                               // harmless assignments
        "b.test = 'b';" +
        "a.test = 'a';" +
        "var a = {}, b = {}, c;";

    // --------------------------------------------------------------------
    // The function name we are checking for inlining.
    // --------------------------------------------------------------------
    final String functionName = "foo";

    // --------------------------------------------------------------------
    // Expected outcome: the reference can be inlined (true).
    // --------------------------------------------------------------------
    final boolean canInline = true;

    // --------------------------------------------------------------------
    // Inlining mode – the existing test suite uses INLINE_BLOCK for these
    // expression‑level checks.
    // --------------------------------------------------------------------
    final InliningMode mode = InliningMode.INLINE_BLOCK;

    // --------------------------------------------------------------------
    // Delegate to the shared helper that performs the actual analysis.
    // --------------------------------------------------------------------
    helperCanInlineReferenceToFunction(canInline, js, functionName, mode);
}
