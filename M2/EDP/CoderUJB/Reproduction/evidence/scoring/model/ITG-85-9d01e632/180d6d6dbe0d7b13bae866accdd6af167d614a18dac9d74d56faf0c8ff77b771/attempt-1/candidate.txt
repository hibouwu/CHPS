public void testBug1864222() {
        // Create a TimeSeries object with some data points
        TimeSeries foo = new TimeSeries("foo", Day.class);
        foo.add(new Day(19, 4, 2005), 1);
        foo.add(new Day(25, 5, 2005), 1);
        foo.add(new Day(28, 5, 2005), 1);
        foo.add(new Day(30, 5, 2005), 1);
        foo.add(new Day(1, 6, 2005), 1);
        foo.add(new Day(3, 6, 2005), 1);
        foo.add(new Day(19, 8, 2005), 1);
        foo.add(new Day(31, 1, 2006), 1);

        // Define the time period that does not overlap with any data points in the series
        Day start = new Day(1, 12, 2005);
        Day end = new Day(18, 1, 2006);

        // Attempt to create a copy of the series within the specified time period
        try {
            TimeSeries bar = foo.createCopy(start, end);
            // If no exception is thrown, the series should be empty
            assertTrue(bar.isEmpty());
        } catch (CloneNotSupportedException e) {
            fail("Unexpected CloneNotSupportedException: " + e.getMessage());
        } catch (IllegalArgumentException e) {
            fail("Unexpected IllegalArgumentException: " + e.getMessage());
        }
    }
