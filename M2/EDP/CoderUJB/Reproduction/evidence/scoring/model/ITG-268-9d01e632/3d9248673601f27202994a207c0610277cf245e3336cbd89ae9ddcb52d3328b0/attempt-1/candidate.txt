@Test
public void supportsBOMinFiles() throws IOException {
        String html = "<html><head><title>BOM test</title></head>"
                + "<body><p>H\u00e9llo \u4e16\u754c</p></body></html>";
        String[] encodings = {"UTF-8", "UTF-16BE", "UTF-16LE"};
        byte[][] boms = {
                {(byte) 0xEF, (byte) 0xBB, (byte) 0xBF},
                {(byte) 0xFE, (byte) 0xFF},
                {(byte) 0xFF, (byte) 0xFE}
        };

        for (int i = 0; i < encodings.length; i++) {
            byte[] content = html.getBytes(encodings[i]);
            byte[] input = new byte[boms[i].length + content.length];
            System.arraycopy(boms[i], 0, input, 0, boms[i].length);
            System.arraycopy(content, 0, input, boms[i].length, content.length);

            org.jsoup.nodes.Document document = org.jsoup.Jsoup.parse(
                    new java.io.ByteArrayInputStream(input), null, "");

            org.junit.Assert.assertEquals(encodings[i], "BOM test", document.title());
            org.junit.Assert.assertEquals(encodings[i], "H\u00e9llo \u4e16\u754c",
                    document.body().text());
            org.junit.Assert.assertFalse(encodings[i],
                    document.text().contains("\uFEFF"));
        }
    }
