@Test
public void supportsBOMinFiles() throws IOException {
        // -----------------------------------------------------------------
        // 1️⃣  Prepare a UTF‑16LE file with a BOM
        // -----------------------------------------------------------------
        // Text that contains characters outside the ASCII range – this will
        // make any mis‑interpretation of the encoding obvious.
        final String original = "Привет, мир! こんにちは世界";

        // UTF‑16LE BOM (0xFF, 0xFE)
        final byte[] bom = new byte[] {(byte)0xFF, (byte)0xFE};

        // Encode the string in UTF‑16LE (the same endianness as the BOM)
        final byte[] encoded = original.getBytes("UTF-16LE");

        // Write BOM + encoded text to a temporary file
        Path tmpFile = Files.createTempFile("jsoup-bom-", ".html");
        try (OutputStream out = Files.newOutputStream(tmpFile)) {
            out.write(bom);          // write the BOM first
            out.write(encoded);      // then the UTF‑16LE payload
        }

        // -----------------------------------------------------------------
        // 2️⃣  Parse the file with Jsoup – let Jsoup auto‑detect the charset
        // -----------------------------------------------------------------
        Document doc = Jsoup.parse(tmpFile.toFile(), null); // charset = null → auto‑detect

        // -----------------------------------------------------------------
        // 3️⃣  Verify that the parsed text matches the original string
        // -----------------------------------------------------------------
        // The body text is the most straightforward way to get the raw
        // character data without any HTML tags interfering.
        String parsed = doc.body().text();

        // If Jsoup had treated the stream as UTF‑8 the result would be a
        // completely different (garbled) string.  The assertion therefore
        // guarantees that the UTF‑16 BOM was honoured.
        org.junit.Assert.assertEquals(
                "The document should be parsed using UTF‑16 (BOM detected)",
                original,
                parsed
        );

        // Clean‑up – delete the temporary file (best‑effort, ignore failures)
        try {
            Files.deleteIfExists(tmpFile);
        } catch (IOException ignored) { }
    }
