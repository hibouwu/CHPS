@Test
public void supportsBOMinFiles() throws IOException {
    // UTF-16 LE BOM: 0xFF 0xFE
    byte[] utf16leBOM = new byte[]{(byte) 0xFF, (byte) 0xFE};
    String utf16leText = "Hello, World!";
    byte[] utf16leBytes = ("Hello, World!").getBytes(StandardCharsets.UTF_16LE);
    byte[] utf16leWithBOM = new byte[utf16leBOM.length + utf16leBytes.length];
    System.arraycopy(utf16leBOM, 0, utf16leWithBOM, 0, utf16leBOM.length);
    System.arraycopy(utf16leBytes, 0, utf16leWithBOM, utf16leBOM.length, utf16leBytes.length);

    // UTF-16 BE BOM: 0xFE 0xFF
    byte[] utf16beBOM = new byte[]{(byte) 0xFE, (byte) 0xFF};
    byte[] utf16beBytes = ("Hello, World!").getBytes(StandardCharsets.UTF_16BE);
    byte[] utf16beWithBOM = new byte[utf16beBOM.length + utf16beBytes.length];
    System.arraycopy(utf16beBOM, 0, utf16beWithBOM, 0, utf16beBOM.length);
    System.arraycopy(utf16beBytes, 0, utf16beWithBOM, utf16beBOM.length, utf16beBytes.length);

    // Test UTF-16 LE with BOM
    String resultUtf16le = DataUtil.parseByteData(new ByteArrayInputStream(utf16leWithBOM), null);
    assertEquals("UTF-16 LE parsing failed", utf16leText, resultUtf16le);

    // Test UTF-16 BE with BOM
    String resultUtf16be = DataUtil.parseByteData(new ByteArrayInputStream(utf16beWithBOM), null);
    assertEquals("UTF-16 BE parsing failed", utf16leText, resultUtf16be);
}
