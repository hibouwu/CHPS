private static NodePointer createChildPointerForStep(
                PropertyOwnerPointer parentPointer, Step step) {
        NodeTest nodeTest = step.getNodeTest();
        if (nodeTest instanceof NodeNameTest) {
            NodeNameTest nodeNameTest = (NodeNameTest) nodeTest;
            QName name = nodeNameTest.getNodeName();
            return parentPointer.getChildPointer(name);
        } else if (nodeTest instanceof NameAttributeTest) {
            NameAttributeTest nameAttributeTest = (NameAttributeTest) nodeTest;
            String localName = nameAttributeTest.getLocalName();
            return parentPointer.getAttributePointer(localName);
        }
        return null;
    }
