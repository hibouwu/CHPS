    private static NodePointer createChildPointerForStep(
                PropertyOwnerPointer parentPointer, Step step) {
        NodePointer pointer = null;
        NodeTest test = step.getNodeTest();
        if (test instanceof NodeNameTest) {
            pointer = parentPointer.getChild(
                    ((NodeNameTest) test).getNodeName(),
                    step.getNodeTest().getNamespaceURI());
        } else if (test instanceof NameAttributeTest) {
            pointer = parentPointer.getAttribute(
                    ((NameAttributeTest) test).getNodeName(),
                    step.getNodeTest().getNamespaceURI());
        }
        return pointer;
    }
