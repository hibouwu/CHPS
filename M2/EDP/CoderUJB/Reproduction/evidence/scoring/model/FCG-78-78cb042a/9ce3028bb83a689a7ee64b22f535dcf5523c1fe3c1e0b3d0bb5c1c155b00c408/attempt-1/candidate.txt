    private static NodePointer createChildPointerForStep(
                PropertyOwnerPointer parentPointer, Step step) {
        NodePointer pointer = null;
        if (step.getNodeTest() instanceof NodeNameTest) {
            pointer = parentPointer.getChildPointer(
                    ((NodeNameTest) step.getNodeTest()).getNodeName());
        } else if (step.getNodeTest() instanceof NameAttributeTest) {
            pointer = parentPointer.getAttributePointer(
                    ((NameAttributeTest) step.getNodeTest()).getNodeName());
        } else {
            throw new JXPathException("Unsupported node test: " +
                    step.getNodeTest());
        }
        return pointer;
    }
