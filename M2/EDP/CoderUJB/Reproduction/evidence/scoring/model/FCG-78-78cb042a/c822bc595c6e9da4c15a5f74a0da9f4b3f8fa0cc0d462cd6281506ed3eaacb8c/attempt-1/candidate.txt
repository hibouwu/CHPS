private static NodePointer createChildPointerForStep(PropertyOwnerPointer parentPointer, Step step) {
    // Get the node test from the step
    NodeTest nodeTest = step.getNodeTest();

    // Check if the node test is a NodeNameTest
    if (nodeTest instanceof NodeNameTest) {
        NodeNameTest nodeNameTest = (NodeNameTest) nodeTest;
        QName nodeName = nodeNameTest.getNodeName();
        
        // Create a child pointer using the property name from the node name
        return parentPointer.getPropertyPointer(nodeName.getName());
    }
    
    // Check if the node test is a NameAttributeTest
    if (nodeTest instanceof NameAttributeTest) {
        NameAttributeTest nameAttributeTest = (NameAttributeTest) nodeTest;
        QName attributeName = nameAttributeTest.getAttributeName();
        
        // Create a child pointer for the attribute
        return parentPointer.getAttributePointer(attributeName);
    }

    // If the node test is not recognized, return null or handle it appropriately
    throw new JXPathException("Unsupported node test type: " + nodeTest.getClass().getName());
}
