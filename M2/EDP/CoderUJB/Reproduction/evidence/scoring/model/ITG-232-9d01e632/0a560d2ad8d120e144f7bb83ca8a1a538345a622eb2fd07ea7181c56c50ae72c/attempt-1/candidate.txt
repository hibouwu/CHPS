public void testCodec112() {
    final byte[] binaryData = new byte[] { 'f', 'o', 'o' };
    final byte[] expected = new byte[] { 'Z', 'm', '9', 'v' };

    // Unchunked output is exactly 4 bytes; no chunk separator should be
    // included when checking maxResultSize.
    final byte[] actual = Base64.encodeBase64(binaryData, false, false, 4);

    assertTrue("Unchunked Base64 encoding should succeed at the exact size limit",
            java.util.Arrays.equals(expected, actual));
}
