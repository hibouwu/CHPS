public void testCodec112() {
        // --------------------------------------------------------------------
        // 1️⃣  Prepare a deterministic input payload
        // --------------------------------------------------------------------
        final byte[] binaryData = "Codec‑CODEC‑112 test payload".getBytes(StandardCharsets.UTF_8);

        // --------------------------------------------------------------------
        // 2️⃣  Choose a maxResultSize that is definitely large enough.
        //    The Base64‑encoded length of the above string is 40 bytes.
        // --------------------------------------------------------------------
        final int maxResultSize = 100;   // any value >= 40 should be valid

        // --------------------------------------------------------------------
        // 3️⃣  Call the method under test – it must NOT throw IllegalArgumentException
        // --------------------------------------------------------------------
        byte[] encoded = null;
        try {
            encoded = Base64.encodeBase64(binaryData, /* isChunked */ false,
                                         /* urlSafe   */ false, maxResultSize);
        } catch (IllegalArgumentException e) {
            fail("Method threw IllegalArgumentException for a valid maxResultSize: " + e.getMessage());
        }

        // --------------------------------------------------------------------
        // 4️⃣  Basic sanity checks on the result
        // --------------------------------------------------------------------
        assertNotNull("Encoded result must not be null", encoded);
        assertTrue("Encoded length must not exceed the supplied maxResultSize",
                   encoded.length <= maxResultSize);

        // --------------------------------------------------------------------
        // 5️⃣  Verify that the encoding itself is correct (optional but useful)
        // --------------------------------------------------------------------
        String expected = Base64.encodeBase64String(binaryData);
        String actual   = new String(encoded, StandardCharsets.UTF_8);
        assertEquals("Base64 encoding does not match the expected value", expected, actual);
    }
