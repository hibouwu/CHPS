        // --------------------------------------------------------------------

        final int maxResultSize = 100;   // any value >= 40 should be valid

        // --------------------------------------------------------------------
        // 3️⃣  Call the method under test – it must NOT throw IllegalArgumentException
        // --------------------------------------------------------------------
        byte[] encoded = null;
        try {
            encoded = Base64.encodeBase64(binaryData, /* isChunked */ false,
                                         /* urlSafe   */ false, maxResultSize);

            fail("Method threw IllegalArgumentException for a valid maxResultSize: " + e.getMessage());
        }

        // --------------------------------------------------------------------
        // 4️⃣  Basic sanity checks on the result

        // 5️⃣  Verify that the encoding itself is correct (optional but useful)
        // --------------------------------------------------------------------

        assertEquals("Base64 encoding does not match the expected value", expected, actual);
    }
}
