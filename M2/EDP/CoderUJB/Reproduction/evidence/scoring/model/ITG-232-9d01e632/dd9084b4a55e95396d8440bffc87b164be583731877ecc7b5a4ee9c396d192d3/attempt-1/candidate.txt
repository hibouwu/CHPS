public void testCodec112() {
        // Test data
        byte[] binaryData = "Hello, World!".getBytes();
        boolean isChunked = false;
        boolean urlSafe = true; // or false, depending on what you want to test
        int maxResultSize = 100; // A valid maxResultSize that should be larger than the encoded result size

        try {
            // Call the method with the test data
            byte[] encodedData = Base64.encodeBase64(binaryData, isChunked, urlSafe, maxResultSize);

            // If no exception is thrown, the test passes
            assertNotNull(encodedData);
            assertTrue("Encoded data should not be empty", encodedData.length > 0);
        } catch (IllegalArgumentException iae) {
            // If an IllegalArgumentException is thrown, the test fails
            fail("Should not throw IllegalArgumentException when isChunked is false and maxResultSize is valid");
        }
    }
