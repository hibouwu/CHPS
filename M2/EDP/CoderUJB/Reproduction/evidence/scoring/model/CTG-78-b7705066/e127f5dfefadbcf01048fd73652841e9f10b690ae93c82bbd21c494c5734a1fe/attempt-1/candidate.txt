@Test
public void testReadingOfFirstStoredEntry() throws Exception {
    // Create a temporary ZIP file with a stored entry
    File tempZipFile = File.createTempFile("test", ".zip");
    tempZipFile.deleteOnExit();

    // Prepare some test data
    String testData = "This is a test stored entry.";
    byte[] testDataBytes = testData.getBytes(Charset.forName("UTF-8"));

    // Create a ZIP file and add a stored entry
    try (ZipArchiveOutputStream zos = new ZipArchiveOutputStream(new FileOutputStream(tempZipFile))) {
        ZipArchiveEntry entry = new ZipArchiveEntry("test.txt");
        entry.setMethod(ZipEntry.STORED);
        entry.setSize(testDataBytes.length);
        CRC32 crc32 = new CRC32();
        crc32.update(testDataBytes);
        entry.setCrc(crc32.getValue());
        zos.putArchiveEntry(entry);
        zos.write(testDataBytes);
        zos.closeArchiveEntry();
    }

    // Open the ZIP file using ZipFile class
    try (ZipFile zipFile = new ZipFile(tempZipFile)) {
        Enumeration<ZipArchiveEntry> entries = zipFile.getEntries();
        assertTrue("ZIP file should contain at least one entry", entries.hasMoreElements());

        // Get the first entry
        ZipArchiveEntry firstEntry = entries.nextElement();
        assertEquals("Entry name should match", "test.txt", firstEntry.getName());
        assertEquals("Entry method should be STORED", ZipEntry.STORED, firstEntry.getMethod());
        assertEquals("Entry size should match", testDataBytes.length, firstEntry.getSize());

        // Read the contents of the first entry
        try (InputStream inputStream = zipFile.getInputStream(firstEntry)) {
            byte[] buffer = new byte[(int) firstEntry.getSize()];
            int bytesRead = inputStream.read(buffer);
            assertEquals("Number of bytes read should match entry size", testDataBytes.length, bytesRead);
            assertArrayEquals("Content of the entry should match", testDataBytes, buffer);
        }
    }
}
