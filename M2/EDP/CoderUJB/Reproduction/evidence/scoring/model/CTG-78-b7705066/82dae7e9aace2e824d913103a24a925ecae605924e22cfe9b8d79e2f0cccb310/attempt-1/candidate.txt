    @Test
    public void testReadingOfFirstStoredEntry() throws Exception {
        final File f = getFile("COMPRESS-264.zip");
        try (final ZipFile zf = new ZipFile(f)) {
            final ZipArchiveEntry ze = zf.getEntry("test.txt");
            try (final InputStream is = zf.getInputStream(ze)) {
                final byte[] b = new byte[1024];
                final int r = is.read(b);
                assertEquals(1024, r);
                assertEquals(1024, is.available());
                assertEquals(1024, is.read(b));
                assertEquals(0, is.available());
                assertEquals(-1, is.read(b));
            }
        }
    }
