    /**
     * Test case for
     * <a href="https://issues.apache.org/jira/browse/COMPRESS-264"
     * >COMPRESS-264</a>.
     */
    @Test
    public void testReadingOfFirstStoredEntry() throws Exception {
        final File zipFile = File.createTempFile("test", ".zip");
        zipFile.deleteOnExit();

        final byte[] data = "Hello stored entry".getBytes("UTF-8");
        CRC32 crc = new CRC32();
        crc.update(data);
        long checksum = crc.getValue();

        try (final org.apache.commons.compress.archivers.zip.ZipArchiveOutputStream zos =
                new org.apache.commons.compress.archivers.zip.ZipArchiveOutputStream(zipFile)) {
            ZipArchiveEntry storedEntry = new ZipArchiveEntry("first.txt");
            storedEntry.setMethod(ZipEntry.STORED);
            storedEntry.setSize(data.length);
            storedEntry.setCrc(checksum);
            zos.putArchiveEntry(storedEntry);
            zos.write(data);
            zos.closeArchiveEntry();

            ZipArchiveEntry deflatedEntry = new ZipArchiveEntry("second.txt");
            zos.putArchiveEntry(deflatedEntry);
            zos.write("deflated".getBytes("UTF-8"));
            zos.closeArchiveEntry();
        }

        try (final ZipFile zf = new ZipFile(zipFile)) {
            ZipArchiveEntry entry = zf.getEntry("first.txt");
            assertNotNull("First entry should be present", entry);
            try (InputStream is = zf.getInputStream(entry)) {
                byte[] read = IOUtils.toByteArray(is);
                assertArrayEquals("Content of stored entry should match", data, read);
            }
        }
    }
