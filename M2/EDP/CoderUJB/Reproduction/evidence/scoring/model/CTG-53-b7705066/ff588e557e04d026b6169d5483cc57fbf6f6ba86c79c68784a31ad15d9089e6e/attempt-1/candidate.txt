@Test
public void testNullCommentEqualsEmptyComment() {
        // Two entries with the same name – the only differing attribute will be the comment.
        ZipArchiveEntry entryWithNullComment = new ZipArchiveEntry("sample.txt");
        ZipArchiveEntry entryWithEmptyComment = new ZipArchiveEntry("sample.txt");

        // By default the comment of a ZipEntry is {@code null}.
        assertNull("Default comment should be null", entryWithNullComment.getComment());

        // Explicitly set an empty comment on the second entry.
        entryWithEmptyComment.setComment("");
        assertEquals("Explicit empty comment should be \"\"", "", entryWithEmptyComment.getComment());

        // The two entries must be equal and must have identical hash codes.
        assertEquals("Entries with null and empty comment should be equal",
                     entryWithNullComment, entryWithEmptyComment);
        assertEquals("Hash codes must match for equal entries",
                     entryWithNullComment.hashCode(), entryWithEmptyComment.hashCode());

        // Changing the comment to a non‑empty value must break equality.
        entryWithEmptyComment.setComment("non‑empty comment");
        assertNotEquals("Entries should no longer be equal after setting a non‑empty comment",
                        entryWithNullComment, entryWithEmptyComment);
    }
