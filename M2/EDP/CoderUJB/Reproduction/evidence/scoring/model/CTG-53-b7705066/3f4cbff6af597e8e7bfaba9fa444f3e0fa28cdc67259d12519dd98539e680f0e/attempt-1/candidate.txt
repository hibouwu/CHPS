@Test
public void testNullCommentEqualsEmptyComment() {
        final ZipArchiveEntry entryWithNullComment = new ZipArchiveEntry("test.txt");
        // Explicitly set null comment (default is null, but for clarity)
        entryWithNullComment.setComment(null);

        final ZipArchiveEntry entryWithEmptyComment = new ZipArchiveEntry("test.txt");
        entryWithEmptyComment.setComment("");

        assertEquals("Entries should be equal when one has null comment and the other empty comment",
                entryWithNullComment, entryWithEmptyComment);
        assertEquals("Hash codes should match for equal entries",
                entryWithNullComment.hashCode(), entryWithEmptyComment.hashCode());
    }
