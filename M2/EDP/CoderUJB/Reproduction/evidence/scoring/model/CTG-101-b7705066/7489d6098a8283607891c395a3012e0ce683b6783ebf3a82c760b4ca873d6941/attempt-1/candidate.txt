public void testKeywords() throws Exception {
    // Test with a JSON string containing the keywords
    String json = "{\"key1\": null, \"key2\": true, \"key3\": false}";

    // Create a JsonFactory instance
    JsonFactory factory = new JsonFactory();

    // Create a JsonParser instance from the JSON string
    try (JsonParser parser = factory.createParser(json)) {
        // Verify the first token is START_OBJECT
        assertEquals(JsonToken.START_OBJECT, parser.nextToken());

        // Verify the first field name and value
        assertEquals("key1", parser.nextFieldName());
        assertEquals(JsonToken.VALUE_NULL, parser.nextToken());

        // Verify the second field name and value
        assertEquals("key2", parser.nextFieldName());
        assertEquals(JsonToken.VALUE_TRUE, parser.nextToken());

        // Verify the third field name and value
        assertEquals("key3", parser.nextFieldName());
        assertEquals(JsonToken.VALUE_FALSE, parser.nextToken());

        // Verify the last token is END_OBJECT
        assertEquals(JsonToken.END_OBJECT, parser.nextToken());

        // Ensure there are no more tokens
        assertNull(parser.nextToken());
    }

    // Test with a JSON string containing the keywords in different order
    json = "{\"key2\": true, \"key3\": false, \"key1\": null}";

    // Create a JsonParser instance from the JSON string
    try (JsonParser parser = factory.createParser(json)) {
        // Verify the first token is START_OBJECT
        assertEquals(JsonToken.START_OBJECT, parser.nextToken());

        // Verify the first field name and value
        assertEquals("key2", parser.nextFieldName());
        assertEquals(JsonToken.VALUE_TRUE, parser.nextToken());

        // Verify the second field name and value
        assertEquals("key3", parser.nextFieldName());
        assertEquals(JsonToken.VALUE_FALSE, parser.nextToken());

        // Verify the third field name and value
        assertEquals("key1", parser.nextFieldName());
        assertEquals(JsonToken.VALUE_NULL, parser.nextToken());

        // Verify the last token is END_OBJECT
        assertEquals(JsonToken.END_OBJECT, parser.nextToken());

        // Ensure there are no more tokens
        assertNull(parser.nextToken());
    }

    // Test with a JSON string containing the keywords as standalone values
    json = "[null, true, false]";

    // Create a JsonParser instance from the JSON string
    try (JsonParser parser = factory.createParser(json)) {
        // Verify the first token is START_ARRAY
        assertEquals(JsonToken.START_ARRAY, parser.nextToken());

        // Verify the first value
        assertEquals(JsonToken.VALUE_NULL, parser.nextToken());

        // Verify the second value
        assertEquals(JsonToken.VALUE_TRUE, parser.nextToken());

        // Verify the third value
        assertEquals(JsonToken.VALUE_FALSE, parser.nextToken());

        // Verify the last token is END_ARRAY
        assertEquals(JsonToken.END_ARRAY, parser.nextToken());

        // Ensure there are no more tokens
        assertNull(parser.nextToken());
    }
}
