public void testKeywords() throws Exception {
    final JsonFactory factory = new JsonFactory();
    final String[] documents = {
        " null \t true \r\n false ",
        "[null,true,false]",
        "{\"a\":null,\"b\":true,\"c\":false}",
        "{\"values\":[null,{\"flag\":true},false]}"
    };
    final JsonToken[][] expectedTokens = {
        {
            JsonToken.VALUE_NULL,
            JsonToken.VALUE_TRUE,
            JsonToken.VALUE_FALSE
        },
        {
            JsonToken.START_ARRAY,
            JsonToken.VALUE_NULL,
            JsonToken.VALUE_TRUE,
            JsonToken.VALUE_FALSE,
            JsonToken.END_ARRAY
        },
        {
            JsonToken.START_OBJECT,
            JsonToken.FIELD_NAME, JsonToken.VALUE_NULL,
            JsonToken.FIELD_NAME, JsonToken.VALUE_TRUE,
            JsonToken.FIELD_NAME, JsonToken.VALUE_FALSE,
            JsonToken.END_OBJECT
        },
        {
            JsonToken.START_OBJECT,
            JsonToken.FIELD_NAME,
            JsonToken.START_ARRAY,
            JsonToken.VALUE_NULL,
            JsonToken.START_OBJECT,
            JsonToken.FIELD_NAME, JsonToken.VALUE_TRUE,
            JsonToken.END_OBJECT,
            JsonToken.VALUE_FALSE,
            JsonToken.END_ARRAY,
            JsonToken.END_OBJECT
        }
    };

    for (int mode = 0; mode < 3; ++mode) {
        for (int i = 0; i < documents.length; ++i) {
            final JsonParser parser;
            switch (mode) {
            case 0:
                parser = factory.createParser(documents[i]);
                break;
            case 1:
                parser = factory.createParser(new StringReader(documents[i]));
                break;
            default:
                parser = factory.createParser(new ByteArrayInputStream(
                        documents[i].getBytes("UTF-8")));
                break;
            }

            try (JsonParser p = parser) {
                for (JsonToken expected : expectedTokens[i]) {
                    assertEquals(expected, p.nextToken());
                    assertEquals(expected, p.getCurrentToken());

                    switch (expected) {
                    case VALUE_NULL:
                        assertEquals("null", p.getText());
                        break;
                    case VALUE_TRUE:
                        assertEquals("true", p.getText());
                        assertTrue(p.getBooleanValue());
                        break;
                    case VALUE_FALSE:
                        assertEquals("false", p.getText());
                        assertFalse(p.getBooleanValue());
                        break;
                    default:
                        break;
                    }
                }
                assertNull(p.nextToken());
            }
        }
    }
}
