/**
 * Some globals should not be aliased because they have special meaning
 * within the language (like arguments).
 */
public void testUnaliasable() {
  // Repeat enough times that usage frequency does not prevent aliasing.
  StringBuilder source = new StringBuilder();
  for (int i = 0; i < 20; i++) {
    source.append("arguments;");
    source.append("eval('0');");
  }

  testSame(source.toString());
}
