private static void appendOption(final StringBuffer buff, 
                                 final Option option, 
                                 final boolean required,
                                 final HelpFormatter formatter)
{
    if (!required)
    {
        buff.append("[");
    }

    if (option.getOpt() != null)
    {
        buff.append(formatter.defaultOptPrefix).append(option.getOpt());
    }
    else if (option.getLongOpt() != null)
    {
        buff.append(formatter.defaultLongOptPrefix).append(option.getLongOpt());
    }
    else
    {
        // Handle the case where neither short nor long option is present
        throw new IllegalArgumentException("Option must have either a short or a long name.");
    }

    // if the Option has a value
    if (option.hasArg())
    {
        String argName = option.getArgName();
        if (argName == null)
        {
            argName = formatter.defaultArgName; // Use default argument name if none is specified
        }
        buff.append(" <").append(argName).append(">");
    }

    // if the Option is not a required option
    if (!required)
    {
        buff.append("]");
    }
}
