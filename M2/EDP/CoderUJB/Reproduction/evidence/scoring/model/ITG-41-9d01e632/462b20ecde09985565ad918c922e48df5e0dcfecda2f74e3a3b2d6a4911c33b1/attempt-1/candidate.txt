public void testGenericSignature1195() throws Exception
    {
        // -----------------------------------------------------------------
        // 1. Build the JavaType for AtomicReference<String>
        // -----------------------------------------------------------------
        TypeFactory tf = TypeFactory.defaultInstance();
        JavaType type = tf.constructParametricType(AtomicReference.class, String.class);

        // -----------------------------------------------------------------
        // 2. Obtain the generic signature
        // -----------------------------------------------------------------
        String signature = type.getGenericSignature();

        // -----------------------------------------------------------------
        // 3. Expected signature (the method‑type prefix “()” is added by
        //    JavaType#getGenericSignature for consistency with method
        //    signatures – see the issue discussion)
        // -----------------------------------------------------------------
        String expected = "()Ljava/util/concurrent/atomic/AtomicReference<Ljava/lang/String;>;";

        // -----------------------------------------------------------------
        // 4. Verify that the signature is exactly as expected
        // -----------------------------------------------------------------
        assertEquals("Generic signature is malformed – missing closing '>'",
                     expected, signature);
    }
