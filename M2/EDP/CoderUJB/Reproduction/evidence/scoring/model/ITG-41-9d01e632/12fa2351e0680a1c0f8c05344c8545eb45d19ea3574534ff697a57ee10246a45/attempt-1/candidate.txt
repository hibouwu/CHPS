public void testGenericSignature1195() throws Exception {
        // Create an ObjectMapper instance
        ObjectMapper objectMapper = new ObjectMapper();
        
        // Get the TypeFactory from the ObjectMapper
        TypeFactory typeFactory = objectMapper.getTypeFactory();
        
        // Construct a JavaType for AtomicReference<String>
        JavaType atomicReferenceType = typeFactory.constructParametricType(AtomicReference.class, String.class);
        
        // Get the generic signature of the JavaType
        String genericSignature = atomicReferenceType.getGenericSignature();
        
        // Expected generic signature
        String expectedSignature = "()Ljava/util/concurrent/atomic/AtomicReference<Ljava/lang/String;>;";

        // Assert that the generated generic signature matches the expected signature
        assertEquals("The generic signature is incorrect", expectedSignature, genericSignature);
    }
