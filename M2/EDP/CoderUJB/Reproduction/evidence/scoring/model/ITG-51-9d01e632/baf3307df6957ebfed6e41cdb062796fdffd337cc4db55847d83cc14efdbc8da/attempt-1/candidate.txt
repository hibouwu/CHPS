public void testEmptyNodeSetOperations() {
    try {
        javax.xml.parsers.DocumentBuilderFactory factory =
                javax.xml.parsers.DocumentBuilderFactory.newInstance();
        org.w3c.dom.Document document = factory.newDocumentBuilder().parse(
                new org.xml.sax.InputSource(
                        new java.io.StringReader("<doc/>")));

        org.apache.commons.jxpath.JXPathContext context =
                org.apache.commons.jxpath.JXPathContext.newContext(document);

        String[] expressions = {
            // An empty node-set compared with a number, in either order.
            "2.0 > child1",
            "2.0 >= child1",
            "2.0 < child1",
            "2.0 <= child1",
            "2.0 = child1",
            "2.0 != child1",
            "child1 > 2.0",
            "child1 >= 2.0",
            "child1 < 2.0",
            "child1 <= 2.0",
            "child1 = 2.0",
            "child1 != 2.0",

            // Comparing two empty node-sets must also return false.
            "child1 > child2",
            "child1 >= child2",
            "child1 < child2",
            "child1 <= child2",
            "child1 = child2",
            "child1 != child2"
        };

        for (int i = 0; i < expressions.length; i++) {
            assertEquals(
                    "Expected false for: " + expressions[i],
                    Boolean.FALSE,
                    context.getValue(expressions[i], Boolean.class));
        }
    } catch (Exception e) {
        throw new AssertionError("Failed to test empty node-set comparisons", e);
    }
}
