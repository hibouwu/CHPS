public long readBits(final int count) throws IOException {
        if (count <= 0 || count > MAXIMUM_CACHE_SIZE) {
            throw new IllegalArgumentException("Count must be between 1 and " + MAXIMUM_CACHE_SIZE);
        }

        if (!ensureCache(count)) {
            return -1;
        }

        long result = bitsCached & MASKS[count];
        bitsCached >>>= count;
        bitsCachedSize -= count;

        return result;
    }
