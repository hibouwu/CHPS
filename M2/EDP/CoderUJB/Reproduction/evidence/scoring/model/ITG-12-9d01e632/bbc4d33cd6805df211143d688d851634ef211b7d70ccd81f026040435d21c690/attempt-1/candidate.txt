public void testEscapeHtmlHighUnicode() throws java.io.UnsupportedEncodingException {
        // UTF‑8 bytes for U+1D362
        byte[] data = new byte[] {
                (byte) 0xF0, (byte) 0x9D, (byte) 0x8D, (byte) 0xA2
        };

        // Build a Java String that contains the single supplementary character
        String original = new String(data, "UTF-8");

        // Escape it
        String escaped = StringEscapeUtils.escapeHtml(original);

        // Expected HTML numeric entity (decimal)
        String expected = "&#119650;";

        assertEquals("HTML escaping of a supplementary Unicode character is wrong",
                     expected, escaped);
    }
