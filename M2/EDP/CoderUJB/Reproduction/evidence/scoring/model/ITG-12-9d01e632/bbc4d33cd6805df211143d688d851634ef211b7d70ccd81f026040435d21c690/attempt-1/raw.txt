**JUnit 4 test case for LANG‑480**

```java
package org.apache.commons.lang;

import static org.junit.Assert.assertEquals;

import org.junit.Test;

/**
 * Test that {@link StringEscapeUtils#escapeHtml(String)} correctly
 * escapes Unicode code‑points that are represented by a surrogate pair
 * (i.e. code‑points > U+00FFFF).
 *
 * <p>Issue: LANG‑480 – “StringEscapeUtils.escapeHtml incorrectly converts
 * unicode characters above U+00FFFF into 2 characters”.</p>
 */
public class StringEscapeUtilsTest {

    /**
     * The character used in the original bug report is
     * <code>U+1D362</code> (COUNTING ROD UNIT DIGIT THREE).
     * Its UTF‑8 byte sequence is {@code F0 9D 8D A2}.
     *
     * The expected HTML‑escaped form is the decimal numeric entity
     * {@code &#119650;}.  The buggy implementation produced
     * {@code &#55348;&#57186;} (the two surrogate halves).
     */
    @Test
    public void testEscapeHtmlHighUnicode() throws java.io.UnsupportedEncodingException {
        // UTF‑8 bytes for U+1D362
        byte[] data = new byte[] {
                (byte) 0xF0, (byte) 0x9D, (byte) 0x8D, (byte) 0xA2
        };

        // Build a Java String that contains the single supplementary character
        String original = new String(data, "UTF-8");

        // Escape it
        String escaped = StringEscapeUtils.escapeHtml(original);

        // Expected HTML numeric entity (decimal)
        String expected = "&#119650;";

        assertEquals("HTML escaping of a supplementary Unicode character is wrong",
                     expected, escaped);
    }
}
```
