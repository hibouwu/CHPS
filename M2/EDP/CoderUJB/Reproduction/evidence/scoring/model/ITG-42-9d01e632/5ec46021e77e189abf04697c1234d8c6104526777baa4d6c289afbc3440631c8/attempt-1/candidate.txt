public void testEqualsAfterSerializationOfDerivedClass() throws IOException, ClassNotFoundException {
        class MultiKey2<A, B> extends MultiKey {
            private static final long serialVersionUID = 1928896152249821416L;

            public MultiKey2(A key1, B key2) {
                super(key1, key2);
            }

            public A getFirst() {
                return (A) getKey(0);
            }

            public B getSecond() {
                return (B) getKey(1);
            }
        }

        MultiKey2<String, String> original = new MultiKey2<>("foo", "bar");
        MultiKey2<String, String> deserialized = serializeAndDeserialize(original);

        // Check if the original and deserialized objects are equal
        assertTrue("Original and deserialized objects should be equal", original.equals(deserialized));
        // Check if the hash codes are equal
        assertEquals("Hash codes of original and deserialized objects should be equal", original.hashCode(), deserialized.hashCode());
    }
