public void testEqualsAfterSerializationOfDerivedClass()
            throws IOException, ClassNotFoundException {

        // -----------------------------------------------------------------
        // 1. Create the original object (a subclass of MultiKey)
        // -----------------------------------------------------------------
        MyMultiKey<String, Integer> original = new MyMultiKey<>("alpha", 42);

        // -----------------------------------------------------------------
        // 2. Serialise it to a byte array
        // -----------------------------------------------------------------
        byte[] serialized;
        try (ByteArrayOutputStream bos = new ByteArrayOutputStream();
             ObjectOutputStream out = new ObjectOutputStream(bos)) {
            out.writeObject(original);
            out.flush();
            serialized = bos.toByteArray();
        }

        // -----------------------------------------------------------------
        // 3. Deserialise the byte array back into an object
        // -----------------------------------------------------------------
        MyMultiKey<String, Integer> deserialized;
        try (ByteArrayInputStream bis = new ByteArrayInputStream(serialized);
             ObjectInputStream in = new ObjectInputStream(bis)) {
            @SuppressWarnings("unchecked")
            MyMultiKey<String, Integer> tmp = (MyMultiKey<String, Integer>) in.readObject();
            deserialized = tmp;
        }

        // -----------------------------------------------------------------
        // 4. Verify that equals() (and hashCode()) are preserved
        // -----------------------------------------------------------------
        assertTrue("Deserialized object must be equal to the original", original.equals(deserialized));
        assertEquals("Hash codes must be identical after serialization round‑trip",
                original.hashCode(), deserialized.hashCode());
    }
