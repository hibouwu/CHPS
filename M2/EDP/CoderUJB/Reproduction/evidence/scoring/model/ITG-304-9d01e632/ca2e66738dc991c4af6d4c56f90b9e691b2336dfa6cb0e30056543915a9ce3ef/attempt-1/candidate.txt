@Test
public void testDontQuoteEuroFirstChar() throws IOException {
        // Prepare the data with CJK characters
        String[] record = {"あ", "い", "う"};

        // Use StringWriter to capture the output
        StringWriter stringWriter = new StringWriter();

        // Create a CSVPrinter with CSVFormat.EXCEL
        try (CSVPrinter csvPrinter = new CSVPrinter(stringWriter, CSVFormat.EXCEL)) {
            // Print the record
            csvPrinter.printRecord(record);
        }

        // Get the output as a string
        String output = stringWriter.toString().trim();

        // Expected output should not have quotes around the first CJK character
        String expectedOutput = "あ,い,う";

        // Assert that the output matches the expected output
        assertEquals(expectedOutput, output, "The first field with CJK characters should not be quoted.");
    }
