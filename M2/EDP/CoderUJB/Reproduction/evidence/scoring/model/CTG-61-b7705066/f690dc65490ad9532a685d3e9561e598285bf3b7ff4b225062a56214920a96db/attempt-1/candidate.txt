@Test
public void testHWRuleEx3() throws EncoderException {
        // Create an instance of Soundex using the default US English mapping
        Soundex soundex = new Soundex();

        // Test cases where consonants from the same code group are separated by 'W' or 'H'
        // Example: "Ashcraft" and "Ashcroft" should both encode to "A261"
        Assert.assertEquals("A261", soundex.encode("Ashcraft"));
        Assert.assertEquals("A261", soundex.encode("Ashcroft"));

        // Another example: "Gough" and "Goff" should both encode to "G100"
        Assert.assertEquals("G100", soundex.encode("Gough"));
        Assert.assertEquals("G100", soundex.encode("Goff"));

        // Example: "Tymczak" and "Tymczaek" should both encode to "T522"
        Assert.assertEquals("T522", soundex.encode("Tymczak"));
        Assert.assertEquals("T522", soundex.encode("Tymczaek"));

        // Example: "Pfister" and "Pfisterer" should both encode to "P236"
        Assert.assertEquals("P236", soundex.encode("Pfister"));
        Assert.assertEquals("P236", soundex.encode("Pfisterer"));
    }
