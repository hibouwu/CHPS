@Test
public void testHWRuleEx3() throws EncoderException {
    // Create a default Soundex instance (US‑English mapping)
    final Soundex soundex = new Soundex();

    // Two strings that differ only by an H between letters of the same group
    final String s1 = "Ashcraft";
    final String s2 = "Ashcroft";

    // Expected Soundex code according to the rule (A‑2‑6‑1, padded to length 4)
    final String expected = "A261";

    // Verify each string individually yields the expected code
    assertEquals("Soundex of \"" + s1 + "\" should be " + expected,
                 expected, soundex.encode(s1));
    assertEquals("Soundex of \"" + s2 + "\" should be " + expected,
                 expected, soundex.encode(s2));

    // And, of course, the two encodings must be identical to each other
    assertEquals("Encodings of \"" + s1 + "\" and \"" + s2 + "\" must match",
                 soundex.encode(s1), soundex.encode(s2));
}
