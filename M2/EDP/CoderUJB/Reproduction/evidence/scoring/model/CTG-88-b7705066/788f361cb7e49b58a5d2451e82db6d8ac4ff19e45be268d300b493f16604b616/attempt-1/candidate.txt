public void testDifferentDurationReturnDifferentObjects() {
        DateTimeFieldType fieldType = DateTimeFieldType.dayOfMonth();
        DurationField durationField1 = new MockDurationField(DurationFieldType.days());
        DurationField durationField2 = new MockDurationField(DurationFieldType.hours());

        UnsupportedDateTimeField field1 = UnsupportedDateTimeField.getInstance(fieldType, durationField1);
        UnsupportedDateTimeField field2 = UnsupportedDateTimeField.getInstance(fieldType, durationField2);

        assertNotNull(field1);
        assertNotNull(field2);
        assertNotSame(field1, field2);
    }
