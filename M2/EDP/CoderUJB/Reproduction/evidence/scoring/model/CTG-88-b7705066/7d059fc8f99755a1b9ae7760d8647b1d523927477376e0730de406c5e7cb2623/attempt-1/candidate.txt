    public void testDifferentDurationReturnDifferentObjects() {
        DateTimeFieldType type = DateTimeFieldType.year();
        DurationField durationField1 = type.getDurationType().getField(type);
        DurationField durationField2 = type.getDurationType().getField(type);
        DateTimeField field1 = UnsupportedDateTimeField.getInstance(type, durationField1);
        DateTimeField field2 = UnsupportedDateTimeField.getInstance(type, durationField2);
