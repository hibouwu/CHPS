public void testCalendar() {
    // Create a date in the Ethiopian calendar using a known date in the Gregorian calendar
    DateTime gregorianDate = new DateTime(2023, 10, 15, 0, 0, 0, 0, DateTimeZone.UTC);
    DateTime ethiopianDate = new DateTime(gregorianDate.getMillis(), ETHIOPIC_UTC);

    // Expected values for the given Gregorian date 2023-10-15 in the Ethiopian calendar
    int expectedEra = 1; // Common Era (CE)
    int expectedYear = 2016; // Year in the Ethiopian calendar
    int expectedMonthOfYear = 13; // Tekemt (13th month in the Ethiopian calendar)
    int expectedDayOfMonth = 5; // Day of the month in the Ethiopian calendar
    int expectedDayOfWeek = DateTimeConstants.SUNDAY; // Day of the week

    // Verify era
    assertEquals("Era", expectedEra, ethiopianDate.getYearOfEra());
    
    // Verify year
    assertEquals("Year", expectedYear, ethiopianDate.getYear());

    // Verify month of year
    assertEquals("Month of Year", expectedMonthOfYear, ethiopianDate.getMonthOfYear());

    // Verify day of month
    assertEquals("Day of Month", expectedDayOfMonth, ethiopianDate.getDayOfMonth());

    // Verify day of week
    assertEquals("Day of Week", expectedDayOfWeek, ethiopianDate.getDayOfWeek());

    // Additional check: Verify that the date conversion is consistent
    DateTime convertedBackToGregorian = ethiopianDate.toDateTime(ISO_UTC);
    assertEquals("Converted back to Gregorian", gregorianDate.withTimeAtStartOfDay(), convertedBackToGregorian.withTimeAtStartOfDay());
}
