public void testCalendar() {
    // Ethiopic 1995-01-01 corresponds to ISO 2002-09-11 (Wednesday).
    long millis = ISO_UTC.getDateTimeMillis(2002, 9, 11, 0);
    int expectedDayOfWeek = DateTimeConstants.WEDNESDAY;

    // Cover ordinary years, leap years and all month/year transitions.
    for (int year = 1995; year <= 2000; year++) {
        for (int month = 1; month <= 13; month++) {
            int daysInMonth = month <= 12 ? 30 : (year % 4 == 3 ? 6 : 5);

            for (int day = 1; day <= daysInMonth; day++) {
                String message = "Ethiopic " + year + "-" + month + "-" + day;

                assertEquals(message, EthiopicChronology.EE,
                        ETHIOPIC_UTC.era().get(millis));
                assertEquals(message, year,
                        ETHIOPIC_UTC.year().get(millis));
                assertEquals(message, month,
                        ETHIOPIC_UTC.monthOfYear().get(millis));
                assertEquals(message, day,
                        ETHIOPIC_UTC.dayOfMonth().get(millis));
                assertEquals(message, expectedDayOfWeek,
                        ETHIOPIC_UTC.dayOfWeek().get(millis));

                assertEquals(message, millis,
                        ETHIOPIC_UTC.getDateTimeMillis(year, month, day, 0));

                millis += MILLIS_PER_DAY;
                expectedDayOfWeek = expectedDayOfWeek == DateTimeConstants.SUNDAY
                        ? DateTimeConstants.MONDAY : expectedDayOfWeek + 1;
            }
        }
    }

    assertEquals(2001, ETHIOPIC_UTC.year().get(millis));
    assertEquals(1, ETHIOPIC_UTC.monthOfYear().get(millis));
    assertEquals(1, ETHIOPIC_UTC.dayOfMonth().get(millis));
}
