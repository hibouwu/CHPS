public void testGoodParse() {
    // Initialize the parser with a simple error manager
    errorManager = new PrintStreamErrorManager(System.err);
    parser = new DepsFileParser(errorManager);

    // Define a sample deps file content with various cases to test
    String fileContent = 
        "// This is a comment\n" +
        "goog.addDependency('file1.js', ['namespace1'], []);\n" +
        "goog.addDependency(\"file2.js\", [], [\"dependency1\", \"dependency2\"]);\n" +
        "goog.addDependency('file3.js', ['namespace2', 'namespace3'], ['dep1', 'dep2']);\n" +
        "goog.addDependency(\"file4.js\", [], []);\n";

    // Parse the file content
    List<DependencyInfo> depInfos = parser.parseFile(SRC_PATH, fileContent);

    // Verify the number of dependencies parsed
    assertEquals(4, depInfos.size());

    // Check each dependency info
    DependencyInfo dep1 = depInfos.get(0);
    assertEquals("file1.js", dep1.getPathRelativeToClosureBase());
    assertEquals(ImmutableList.of("namespace1"), dep1.getProvides());
    assertEquals(EMPTY, dep1.getRequires());

    DependencyInfo dep2 = depInfos.get(1);
    assertEquals("file2.js", dep2.getPathRelativeToClosureBase());
    assertEquals(EMPTY, dep2.getProvides());
    assertEquals(ImmutableList.of("dependency1", "dependency2"), dep2.getRequires());

    DependencyInfo dep3 = depInfos.get(2);
    assertEquals("file3.js", dep3.getPathRelativeToClosureBase());
    assertEquals(ImmutableList.of("namespace2", "namespace3"), dep3.getProvides());
    assertEquals(ImmutableList.of("dep1", "dep2"), dep3.getRequires());

    DependencyInfo dep4 = depInfos.get(3);
    assertEquals("file4.js", dep4.getPathRelativeToClosureBase());
    assertEquals(EMPTY, dep4.getProvides());
    assertEquals(EMPTY, dep4.getRequires());
}
