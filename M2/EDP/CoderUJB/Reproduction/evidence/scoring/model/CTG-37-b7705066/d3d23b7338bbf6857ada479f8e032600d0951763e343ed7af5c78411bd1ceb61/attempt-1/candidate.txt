public void testGoodParse() {
  String contents =
      "// goog.addDependency('ignored.js', ['ignored'], []);\n"
      + "/* A block comment containing a dependency:\n"
      + "goog.addDependency('also_ignored.js', [], ['ignored']);\n"
      + "*/\n"
      + "goog.addDependency('empty.js', [], []);\n"
      + "goog.addDependency(\"provides.js\", ['example.One', \"example.Two\"], []);\n"
      + "goog.addDependency('requires.js', [\"example.Three\"], "
      + "['example.One', \"example.Two\"]); // trailing comment\n";

  List<DependencyInfo> dependencies = parser.parseFile(SRC_PATH, contents);

  assertEquals(0, errorManager.getErrorCount());
  assertEquals(0, errorManager.getWarningCount());
  assertEquals(3, dependencies.size());

  DependencyInfo empty = dependencies.get(0);
  assertEquals("empty.js", empty.getPathRelativeToClosureBase());
  assertEquals(EMPTY, empty.getProvides());
  assertEquals(EMPTY, empty.getRequires());

  DependencyInfo provides = dependencies.get(1);
  assertEquals("provides.js", provides.getPathRelativeToClosureBase());
  assertEquals(
      ImmutableList.of("example.One", "example.Two"), provides.getProvides());
  assertEquals(EMPTY, provides.getRequires());

  DependencyInfo requires = dependencies.get(2);
  assertEquals("requires.js", requires.getPathRelativeToClosureBase());
  assertEquals(ImmutableList.of("example.Three"), requires.getProvides());
  assertEquals(
      ImmutableList.of("example.One", "example.Two"), requires.getRequires());
}
