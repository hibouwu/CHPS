  public void testGoodParse() {
    String contents =
        "// A dependency in a comment must be ignored.\n"
            + "// goog.addDependency('ignored.js', ['ignored.Name'], []);\n"
            + "/* A block comment containing another dependency:\n"
            + "goog.addDependency('also_ignored.js', [], ['ignored.Require']);\n"
            + "*/\n"
            + "goog.addDependency('empty.js', [], []);\n"
            + "goog.addDependency(\"provides.js\", [\"example.One\", 'example.Two'], []);\n"
            + "goog.addDependency('requires.js', [], ['example.One', \"example.Two\"]);\n"
            + "goog.addDependency(\"both.js\", ['example.Three'], [\"example.One\"]);\n";

    List<DependencyInfo> dependencies = parser.parseFile(SRC_PATH, contents);

    assertEquals(0, errorManager.getErrorCount());
    assertEquals(0, errorManager.getWarningCount());
    assertEquals(4, dependencies.size());

    DependencyInfo empty = dependencies.get(0);
    assertEquals("empty.js", empty.getPathRelativeToClosureBase());
    assertEquals(EMPTY, empty.getProvides());
    assertEquals(EMPTY, empty.getRequires());

    DependencyInfo provides = dependencies.get(1);
    assertEquals("provides.js", provides.getPathRelativeToClosureBase());
    assertEquals(ImmutableList.of("example.One", "example.Two"), provides.getProvides());
    assertEquals(EMPTY, provides.getRequires());

    DependencyInfo requires = dependencies.get(2);
    assertEquals("requires.js", requires.getPathRelativeToClosureBase());
    assertEquals(EMPTY, requires.getProvides());
    assertEquals(ImmutableList.of("example.One", "example.Two"), requires.getRequires());

    DependencyInfo both = dependencies.get(3);
    assertEquals("both.js", both.getPathRelativeToClosureBase());
    assertEquals(ImmutableList.of("example.Three"), both.getProvides());
    assertEquals(ImmutableList.of("example.One"), both.getRequires());
  }
