@Override
public void visit(NodeTraversal t, Node n, Node parent) {
    // Skip processing if the current node is the root node of the scope
    if (n == scope.getRootNode()) {
        return;
    }

    // Handle function inputs if the node is a parameter list and its parent is the root node
    if (n.getType() == Token.PARAM_LIST && parent == scope.getRootNode()) {
        handleFunctionInputs(parent);
        return;
    }

    // Attach literal types to the node
    attachLiteralTypes(n);

    // Switch statement to handle specific node types
    switch (n.getType()) {
        case Token.FUNCTION:
            // Define the declared function in the scope
            defineDeclaredFunction(n, parent);
            break;
        case Token.CATCH:
            // Define the catch parameter in the scope
            defineCatch(n, parent);
            break;
        case Token.VAR:
            // Define the variable in the scope
            defineVar(n, parent);
            break;
    }
}
