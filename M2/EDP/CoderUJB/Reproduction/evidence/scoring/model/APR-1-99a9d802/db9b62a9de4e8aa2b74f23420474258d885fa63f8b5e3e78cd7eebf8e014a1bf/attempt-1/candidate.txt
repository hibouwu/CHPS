@Override
public JsonToken nextToken() throws IOException
{
    /*------------------------------------------------------------
     * 1.  First, expose any tokens that have been buffered because
     *     the caller asked for the “path” to a matching value.
     *------------------------------------------------------------*/
    TokenFilterContext ctxt = _exposedContext;
    if (ctxt != null) {
        while (true) {
            JsonToken buffered = ctxt.nextTokenToRead();
            if (buffered != null) {                     // still something to emit
                _currToken = buffered;
                return buffered;
            }

            // No more buffered tokens for this context
            if (ctxt == _headContext) {                 // we have reached the top
                _exposedContext = null;                 // clear flag
                break;                                   // go on to read from delegate
            }

            // Move up one level in the filtered‑context chain
            ctxt = _headContext.findChildOf(ctxt);
            _exposedContext = ctxt;
            if (ctxt == null) {                         // should never happen
                throw _constructError("Unexpected problem: chain of filtered context broken");
            }
        }
    }

    /*------------------------------------------------------------
     * 2.  Pull the next token from the underlying parser.
     *------------------------------------------------------------*/
    JsonToken token = delegate.nextToken();
    if (token == null) {                               // end of input
        return _currToken = null;
    }

    /*------------------------------------------------------------
     * 3.  Decide whether the token should be returned, skipped or
     *     cause buffering of surrounding structure.
     *------------------------------------------------------------*/
    TokenFilter filterToUse;

    switch (token.id()) {

        /*--------------------------------------------------------
         *  START_ARRAY  –  decide whether the array (and possibly
         *  its contents) should be exposed.
         *--------------------------------------------------------*/
        case ID_START_ARRAY:
            filterToUse = _itemFilter;
            if (filterToUse == TokenFilter.INCLUDE_ALL) {
                _headContext = _headContext.createChildArrayContext(filterToUse, true);
                return _currToken = token;
            }
            if (filterToUse == null) {                 // nothing matches – skip whole array
                delegate.skipChildren();
                break;
            }

            // Ask the current context whether the value is to be included
            filterToUse = _headContext.checkValue(filterToUse);
            if (filterToUse == null) {
                delegate.skipChildren();
                break;
            }

            // If the filter does not request full inclusion, ask it to
            // possibly transform the start‑array token.
            if (filterToUse != TokenFilter.INCLUDE_ALL) {
                filterToUse = filterToUse.filterStartArray();
            }
            _itemFilter = filterToUse;

            // Create the new context (the second argument tells whether the
            // start token itself should be emitted)
            boolean includeStart = (filterToUse == TokenFilter.INCLUDE_ALL);
            _headContext = _headContext.createChildArrayContext(filterToUse, includeStart);
            if (includeStart) {
                return _currToken = token;
            }

            // If the caller wants the *path* to the matching values we must
            // buffer the start token until a matching leaf is found.
            if (_includePath) {
                token = _nextTokenWithBuffering(_headContext);
                if (token != null) {
                    _currToken = token;
                    return token;
                }
            }
            break;                                      // continue with normal flow

        /*--------------------------------------------------------
         *  START_OBJECT – analogous to START_ARRAY
         *--------------------------------------------------------*/
        case ID_START_OBJECT:
            filterToUse = _itemFilter;
            if (filterToUse == TokenFilter.INCLUDE_ALL) {
                _headContext = _headContext.createChildObjectContext(filterToUse, true);
                return _currToken = token;
            }
            if (filterToUse == null) {
                delegate.skipChildren();
                break;
            }

            filterToUse = _headContext.checkValue(filterToUse);
            if (filterToUse == null) {
                delegate.skipChildren();
                break;
            }

            if (filterToUse != TokenFilter.INCLUDE_ALL) {
                filterToUse = filterToUse.filterStartObject();
            }
            _itemFilter = filterToUse;

            includeStart = (filterToUse == TokenFilter.INCLUDE_ALL);
            _headContext = _headContext.createChildObjectContext(filterToUse, includeStart);
            if (includeStart) {
                return _currToken = token;
            }

            if (_includePath) {
                token = _nextTokenWithBuffering(_headContext);
                if (token != null) {
                    _currToken = token;
                    return token;
                }
            }
            break;

        /*--------------------------------------------------------
         *  END_ARRAY / END_OBJECT – close the current context.
         *--------------------------------------------------------*/
        case ID_END_ARRAY:
        case ID_END_OBJECT:
            // First, possibly notify the filter that the container is finished
            filterToUse = _headContext.getFilter();
            if (filterToUse != null && filterToUse != TokenFilter.INCLUDE_ALL) {
                filterToUse.filterFinishArray(); // same method works for objects
            }

            // Remember whether the start token was already emitted
            boolean returnEnd = _headContext.isStartHandled();

            // Move up one level
            _headContext = _headContext.getParent();
            _itemFilter = (_headContext == null) ? null : _headContext.getFilter();

            if (returnEnd) {
                return _currToken = token;
            }
            break;

        /*--------------------------------------------------------
         *  FIELD_NAME – decide whether the following value should be
         *  included.  The name token itself is always emitted when the
         *  parent object is part of the path.
         *--------------------------------------------------------*/
        case ID_FIELD_NAME:
            String name = delegate.getCurrentName();   // name of the field we just read
            filterToUse = _headContext.setFieldName(name);

            if (filterToUse == TokenFilter.INCLUDE_ALL) {
                // Whole object is included – we only need to emit the name token
                _itemFilter = filterToUse;
                // If the caller does **not** want the whole path we may need to
                // emit the surrounding START_OBJECT token now.
                if (!_includePath && _headContext.isStartHandled() == false) {
                    // Emit the START_OBJECT that opens this object, then the name
                    JsonToken start = _headContext.nextTokenToRead(); // START_OBJECT
                    _exposedContext = _headContext;                  // remember for later
                    _currToken = start;
                    return start;
                }
                return _currToken = token;
            }

            if (filterToUse == null) {                 // field is completely filtered out
                // Skip the value (which may be a whole container)
                delegate.nextToken();                  // move to the value token
                delegate.skipChildren();               // and skip it
                break;
            }

            // Ask the filter whether this particular property should be kept
            filterToUse = filterToUse.includeProperty(name);
            if (filterToUse == null) {                 // property filtered out
                delegate.nextToken();
                delegate.skipChildren();
                break;
            }

            _itemFilter = filterToUse;

            // If the filter now says “include everything from here on”
            if (filterToUse == TokenFilter.INCLUDE_ALL) {
                if (_includePath) {
                    // The name token itself is enough – the path will be emitted later
                    return _currToken = token;
                }
                // otherwise we fall through to normal handling (the value will be emitted)
            }

            // If we need the path, buffer the start token of the container that
            // holds this field (the START_OBJECT that opened the parent)
            if (_includePath) {
                JsonToken buffered = _nextTokenWithBuffering(_headContext);
                if (buffered != null) {
                    _currToken = buffered;
                    return buffered;
                }
            }
            break;

        /*--------------------------------------------------------
         *  SCALAR values (NUMBER, STRING, BOOLEAN, NULL)
         *--------------------------------------------------------*/
        default:
            filterToUse = _itemFilter;
            if (filterToUse == TokenFilter.INCLUDE_ALL) {
                return _currToken = token;
            }
            if (filterToUse != null) {
                filterToUse = _headContext.checkValue(filterToUse);
                if (filterToUse == TokenFilter.INCLUDE_ALL ||
                    (filterToUse != null && filterToUse.includeValue(delegate))) {
                    return _currToken = token;
                }
            }
            // No match – just skip (nothing to do, we already consumed it)
            break;
    }

    /*------------------------------------------------------------
     * 4.  If we reach this point the token we just read was not
     *     part of the output.  Continue the search recursively.
     *------------------------------------------------------------*/
    return _nextToken2();
}
