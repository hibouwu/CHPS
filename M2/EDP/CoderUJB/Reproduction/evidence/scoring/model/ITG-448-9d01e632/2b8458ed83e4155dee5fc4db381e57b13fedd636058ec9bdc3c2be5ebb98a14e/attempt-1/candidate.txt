public void testFormat() {
        final java.util.TimeZone timeZone = java.util.TimeZone.getTimeZone("UTC");
        final java.util.Locale locale = java.util.Locale.US;
        final java.util.Calendar calendar =
                new java.util.GregorianCalendar(timeZone, locale);
        calendar.clear();
        calendar.set(2003, java.util.Calendar.JULY, 15, 12, 0, 0);

        final String[] patterns = {"y", "yy", "yyy", "yyyy", "yyyyy"};
        final String[] expected = {"2003", "03", "2003", "2003", "02003"};

        for (int i = 0; i < patterns.length; i++) {
            final org.apache.commons.lang3.time.FastDateFormat format =
                    org.apache.commons.lang3.time.FastDateFormat.getInstance(
                            patterns[i], timeZone, locale);
            assertEquals("Incorrect year formatting for pattern " + patterns[i],
                    expected[i], format.format(calendar.getTime()));
        }
    }
