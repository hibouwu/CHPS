    public void testFormat() {
        FastDateFormat fastDateFormat = FastDateFormat.getInstance("Y");
        SimpleDateFormat simpleDateFormat = new SimpleDateFormat("Y");
        
        Date date = new GregorianCalendar(2003, Calendar.JANUARY, 1).getTime();
        
        String fastFormattedDate = fastDateFormat.format(date);
        String simpleFormattedDate = simpleDateFormat.format(date);
        
        assertEquals(fastFormattedDate, simpleFormattedDate);
    }
