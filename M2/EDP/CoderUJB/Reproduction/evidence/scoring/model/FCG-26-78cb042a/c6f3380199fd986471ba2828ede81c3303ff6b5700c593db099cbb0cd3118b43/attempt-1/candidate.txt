public TextNode splitText(int offset) {
        Validate.isTrue(offset >= 0 && offset <= text.length(), "Split offset must be between 0 and the length of the text.");

        String beforeOffset = text.substring(0, offset);
        String afterOffset = text.substring(offset);

        // Update this node's text to be the part before the offset
        this.text = beforeOffset;

        // Create a new text node for the part after the offset
        TextNode newNode = new TextNode(afterOffset, baseUri());

        // Insert the new node as the next sibling
        if (parent() != null) {
            parent().addChildren(siblingIndex() + 1, newNode);
        }

        return newNode;
    }
