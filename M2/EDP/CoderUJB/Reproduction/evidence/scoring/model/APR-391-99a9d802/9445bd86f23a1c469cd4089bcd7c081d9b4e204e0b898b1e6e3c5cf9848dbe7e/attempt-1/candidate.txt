@Override
public double integrate(final FirstOrderDifferentialEquations equations,
                            final double t0, final double[] y0,
                            final double t, final double[] y)
            throws DerivativeException, IntegratorException {

        // -------------------------------------------------------------
        // 1 – basic checks and initialisation
        // -------------------------------------------------------------
        sanityChecks(equations, t0, y0, t, y);
        setEquations(equations);
        resetEvaluations();

        final boolean forward = t > t0;

        // working arrays -------------------------------------------------
        final int stages = c.length + 1;                 // number of RK stages
        if (y != y0) {                                   // copy the initial state if needed
            System.arraycopy(y0, 0, y, 0, y0.length);
        }
        final double[][] yDotK = new double[stages][y0.length];
        final double[] yTmp = new double[y0.length];     // temporary state used during a step

        // -------------------------------------------------------------
        // 2 – interpolator (dense output) handling
        // -------------------------------------------------------------
        final AbstractStepInterpolator interpolator;
        if (requiresDenseOutput() || (!eventsHandlersManager.isEmpty())) {
            // we need a real interpolator that can give the state at any
            // intermediate point; we give it its own copy of the state array
            // (the prototype already knows how to copy itself)
            final RungeKuttaStepInterpolator rki =
                    (RungeKuttaStepInterpolator) prototype.copy();
            rki.reinitialize(this, yTmp.clone(), yDotK, forward);
            interpolator = rki;
        } else {
            interpolator = new DummyStepInterpolator(yTmp.clone(), forward);
        }
        interpolator.storeTime(t0);

        // -------------------------------------------------------------
        // 3 – integration control objects
        // -------------------------------------------------------------
        stepStart = t0;
        double hNew = 0;                 // size proposed for the *next* step
        boolean firstTime = true;       // true only for the very first step
        for (StepHandler handler : stepHandlers) {
            handler.reset();
        }
        CombinedEventsManager manager = addEndTimeChecker(t0, t, eventsHandlersManager);
        boolean lastStep = false;

        // -------------------------------------------------------------
        // 4 – main integration loop
        // -------------------------------------------------------------
        while (!lastStep) {

            // -----------------------------------------------------------------
            // a – shift the interpolator (prepare it for the new step)
            // -----------------------------------------------------------------
            interpolator.shift();

            double error = Double.NaN;          // error of the *accepted* step
            boolean stepAccepted = false;      // will become true when we leave the inner loop

            // -----------------------------------------------------------------
            // b – inner loop: try a step, possibly reject it and recompute
            // -----------------------------------------------------------------

                // -------------------------------------------------------------
                // i – first stage (k0) – compute only if we really need it
                // -------------------------------------------------------------
                if (firstTime || !fsal) {
                    computeDerivatives(stepStart, y, yDotK[0]);
                }

                // -------------------------------------------------------------
                // ii – first step size estimation (only once, at the very start)
                // -------------------------------------------------------------
                if (firstTime) {
                    final double[] scale;
                    if (vecAbsoluteTolerance == null) {
                        scale = new double[y0.length];
                        Arrays.fill(scale, scalAbsoluteTolerance);
                    } else {
                        scale = vecAbsoluteTolerance;
                    }
                    hNew = initializeStep(equations, forward, getOrder(),
                                          scale, stepStart, y, yDotK[0],
                                          yTmp, yDotK[1]);
                    firstTime = false;
                }
