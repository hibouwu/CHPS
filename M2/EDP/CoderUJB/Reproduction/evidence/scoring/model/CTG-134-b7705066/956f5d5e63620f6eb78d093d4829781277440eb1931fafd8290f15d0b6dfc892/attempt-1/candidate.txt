    @Test
    public void testMessageWithCorruptFileName() throws Exception {
        final File f = getFile("COMPRESS-351.zip");
        try (final ZipArchiveInputStream zis = new ZipArchiveInputStream(new FileInputStream(f))) {
            final ZipArchiveEntry ze = zis.getNextZipEntry();
            assertEquals("test.txt", ze.getName());
            assertEquals(0, zis.read(new byte[1024]));
            try {
                zis.getNextZipEntry();
                fail("Should have thrown an exception");
            } catch (final ZipException e) {
                assertEquals("ZipException: Corrupt ZIP file - " +
                             "no end of central directory", e.getMessage());
            }
        }
    }
