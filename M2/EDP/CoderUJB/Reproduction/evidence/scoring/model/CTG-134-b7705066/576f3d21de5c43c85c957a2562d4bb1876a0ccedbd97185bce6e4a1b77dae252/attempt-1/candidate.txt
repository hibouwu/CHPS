@Test
public void testMessageWithCorruptFileName() throws Exception {
        // Construct a zip stream with a local file header that declares a filename length
        // longer than the actual bytes provided.
        final java.io.ByteArrayOutputStream baos = new java.io.ByteArrayOutputStream();
        // Local file header signature (PK\003\004)
        baos.write(new byte[] {0x50, 0x4b, 0x03, 0x04});
        // Version needed to extract (2.0)
        baos.write(new byte[] {0x14, 0x00});
        // General purpose bit flag
        baos.write(new byte[] {0x00, 0x00});
        // Compression method (stored)
        baos.write(new byte[] {0x00, 0x00});
        // Last mod file time & date
        baos.write(new byte[] {0x00, 0x00, 0x00, 0x00});
        // CRC-32
        baos.write(new byte[] {0x00, 0x00, 0x00, 0x00});
        // Compressed size
        baos.write(new byte[] {0x00, 0x00, 0x00, 0x00});
        // Uncompressed size
        baos.write(new byte[] {0x00, 0x00, 0x00, 0x00});
        // File name length = 4 (but we will provide only 2 bytes)
        baos.write(new byte[] {0x04, 0x00});
        // Extra field length = 0
        baos.write(new byte[] {0x00, 0x00});
        // File name bytes (only 2 bytes instead of 4)
        baos.write(new byte[] {0x61, 0x62});
        // No extra field, no file data
        final byte[] zipBytes = baos.toByteArray();

        try (InputStream is = new ByteArrayInputStream(zipBytes);
             ZipArchiveInputStream zis = new ZipArchiveInputStream(is)) {
            zis.getNextZipEntry();
            fail("Expected ZipException due to corrupt file name");
        } catch (ZipException e) {
            // The message should indicate an invalid entry name and not cause NPE
            assertTrue("Exception message should mention invalid entry name",
                       e.getMessage().toLowerCase().contains("invalid entry name"));
        }
    }
