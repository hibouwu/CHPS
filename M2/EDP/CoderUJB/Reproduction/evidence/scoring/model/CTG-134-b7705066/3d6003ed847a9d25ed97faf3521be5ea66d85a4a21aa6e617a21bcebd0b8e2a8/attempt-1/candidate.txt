@Test
public void testMessageWithCorruptFileName() throws Exception {
        // Create a ZIP file with a corrupt file name
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        try (ZipArchiveOutputStream zaos = new ZipArchiveOutputStream(baos)) {
            ZipArchiveEntry entry = new ZipArchiveEntry("validFileName.txt");
            zaos.putArchiveEntry(entry);
            zaos.write("This is a valid file content.".getBytes(StandardCharsets.UTF_8));
            zaos.closeArchiveEntry();

            // Manually corrupt the file name in the local file header
            byte[] zipBytes = baos.toByteArray();
            // The file name starts at offset 30 in the local file header
            // Let's corrupt the first character of the file name
            zipBytes[30] = (byte) 0xFF; // Corrupt the first character

            // Now create a ZipArchiveInputStream from the corrupted bytes
            try (ZipArchiveInputStream zais = new ZipArchiveInputStream(new ByteArrayInputStream(zipBytes))) {
                ZipArchiveEntry corruptedEntry = zais.getNextZipEntry();
                Assert.assertNotNull("Expected an entry but got null", corruptedEntry);

                // Attempt to read the entry data
                byte[] buffer = new byte[1024];
                int bytesRead = zais.read(buffer);
                Assert.assertTrue("Expected to read some bytes but read " + bytesRead, bytesRead > 0);

                // Close the entry
                zais.closeEntry();

                // Check if the next entry is null (since there's only one entry)
                ZipArchiveEntry nextEntry = zais.getNextZipEntry();
                Assert.assertNull("Expected no more entries but found one", nextEntry);
            }
        } catch (IOException e) {
            // If an exception is thrown, it should be due to the corrupt file name
            Assert.assertTrue("Expected an IOException but got " + e.getClass().getName(), e instanceof IOException);
            Assert.assertTrue("Expected a specific error message but got " + e.getMessage(),
                    e.getMessage().contains("invalid code"));
        }
    }
