import com.fasterxml.jackson.dataformat.xml.XmlMapper;
import com.fasterxml.jackson.dataformat.xml.annotation.JacksonXmlElementWrapper;
import com.fasterxml.jackson.dataformat.xml.annotation.JacksonXmlProperty;
import com.fasterxml.jackson.dataformat.xml.annotation.JacksonXmlRootElement;
import org.junit.Test;

import java.util.ArrayList;
import java.util.List;

import static org.junit.Assert.*;

public class NestedUnwrappedLists180Test {

    @JacksonXmlRootElement(localName = "definition")
    public static class Definition {
        @JacksonXmlElementWrapper(useWrapping = false)
        @JacksonXmlProperty(localName = "record")
        public List<Record> records = new ArrayList<Record>();
    }

    public static class Record {
        @JacksonXmlProperty(isAttribute = true)
        public String id;

        @JacksonXmlElementWrapper(useWrapping = false)
        @JacksonXmlProperty(localName = "field")
        public List<Field> fields = new ArrayList<Field>();
    }

    public static class Field {
        public String value;
    }

    @Test
    public void testNestedUnwrappedLists180() throws Exception {
        String xml =
                "<definition>"
              + "  <record id=\"first\">"
              + "    <field><value>a</value></field>"
              + "    <field><value>b</value></field>"
              + "  </record>"
              + "  <record id=\"empty\"/>"
              + "  <record id=\"last\">"
              + "    <field><value>c</value></field>"
              + "    <field><value>d</value></field>"
              + "  </record>"
              + "</definition>";

        Definition result = new XmlMapper().readValue(xml, Definition.class);

        assertNotNull(result);
        assertNotNull(result.records);
        // The missing inner list must not split or merge outer records.
        assertEquals(3, result.records.size());

        Record first = result.records.get(0);
        assertEquals("first", first.id);
        assertEquals(2, first.fields.size());
        assertEquals("a", first.fields.get(0).value);
        assertEquals("b", first.fields.get(1).value);

        Record empty = result.records.get(1);
        assertEquals("empty", empty.id);
        assertNotNull(empty.fields);
        assertTrue(empty.fields.isEmpty());

        Record last = result.records.get(2);
        assertEquals("last", last.id);
        assertEquals(2, last.fields.size());
        assertEquals("c", last.fields.get(0).value);
        assertEquals("d", last.fields.get(1).value);
    }
}
