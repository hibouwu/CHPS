public void testNestedUnwrappedLists180() throws Exception {
        class Models {
            public class Field {
                public String value;
            }

            public class Record {
                public String name;

                @com.fasterxml.jackson.dataformat.xml.annotation.JacksonXmlElementWrapper(useWrapping = false)
                @com.fasterxml.jackson.dataformat.xml.annotation.JacksonXmlProperty(localName = "field")
                public java.util.List<Field> fields;
            }

            public class Definition {
                @com.fasterxml.jackson.dataformat.xml.annotation.JacksonXmlElementWrapper(useWrapping = false)
                @com.fasterxml.jackson.dataformat.xml.annotation.JacksonXmlProperty(localName = "record")
                public java.util.List<Record> records;
            }
        }

        final Models models = new Models();
        com.fasterxml.jackson.databind.module.SimpleModule module =
                new com.fasterxml.jackson.databind.module.SimpleModule();
        module.addValueInstantiator(Models.Definition.class,
                new com.fasterxml.jackson.databind.deser.ValueInstantiator.Base(Models.Definition.class) {
                    @Override
                    public boolean canCreateUsingDefault() {
                        return true;
                    }

                    @Override
                    public Object createUsingDefault(
                            com.fasterxml.jackson.databind.DeserializationContext context) {
                        return models.new Definition();
                    }
                });
        module.addValueInstantiator(Models.Record.class,
                new com.fasterxml.jackson.databind.deser.ValueInstantiator.Base(Models.Record.class) {
                    @Override
                    public boolean canCreateUsingDefault() {
                        return true;
                    }

                    @Override
                    public Object createUsingDefault(
                            com.fasterxml.jackson.databind.DeserializationContext context) {
                        return models.new Record();
                    }
                });
        module.addValueInstantiator(Models.Field.class,
                new com.fasterxml.jackson.databind.deser.ValueInstantiator.Base(Models.Field.class) {
                    @Override
                    public boolean canCreateUsingDefault() {
                        return true;
                    }

                    @Override
                    public Object createUsingDefault(
                            com.fasterxml.jackson.databind.DeserializationContext context) {
                        return models.new Field();
                    }
                });

        com.fasterxml.jackson.dataformat.xml.XmlMapper mapper =
                new com.fasterxml.jackson.dataformat.xml.XmlMapper();
        mapper.registerModule(module);

        String xml = "<definition>"
                + "<record><name>first</name>"
                + "<field><value>a</value></field>"
                + "<field><value>b</value></field></record>"
                + "<record><name>empty</name></record>"
                + "<record><name>last</name>"
                + "<field><value>c</value></field>"
                + "<field><value>d</value></field></record>"
                + "</definition>";

        Models.Definition definition = mapper.readValue(xml, Models.Definition.class);

        assertNotNull(definition.records);
        assertEquals(3, definition.records.size());

        Models.Record first = definition.records.get(0);
        assertEquals("first", first.name);
        assertNotNull(first.fields);
        assertEquals(2, first.fields.size());
        assertEquals("a", first.fields.get(0).value);
        assertEquals("b", first.fields.get(1).value);

        Models.Record empty = definition.records.get(1);
        assertEquals("empty", empty.name);
        assertTrue(empty.fields == null || empty.fields.isEmpty());

        Models.Record last = definition.records.get(2);
        assertEquals("last", last.name);
        assertNotNull(last.fields);
        assertEquals(2, last.fields.size());
        assertEquals("c", last.fields.get(0).value);
        assertEquals("d", last.fields.get(1).value);
    }
