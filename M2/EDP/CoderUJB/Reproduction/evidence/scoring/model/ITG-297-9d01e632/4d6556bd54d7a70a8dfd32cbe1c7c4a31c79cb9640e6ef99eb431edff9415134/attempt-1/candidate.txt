public void testNestedUnwrappedLists180() throws Exception {
        String xml = "<root>" +
                     "  <records>" +
                     "    <record>" +
                     "      <fields/>" +
                     "    </record>" +
                     "  </records>" +
                     "</root>";

        ObjectMapper mapper = new XmlMapper();
        Root root = mapper.readValue(xml, Root.class);

        assertNotNull(root.records);
        assertEquals(1, root.records.size());
        Record record = root.records.get(0);
        assertNotNull(record.fields);
        assertTrue(record.fields.isEmpty());
    }
