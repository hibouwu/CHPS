public void testNestedUnwrappedLists180() throws Exception {
        // Create an instance of Definition with an empty inner list
        Definition definition = new Definition();
        Record record = new Record();
        record.setFields(new ArrayList<>()); // Empty list of fields
        definition.getRecords().add(record);

        // Serialize the Definition object to XML
        XmlMapper xmlMapper = new XmlMapper();
        String xml = xmlMapper.writeValueAsString(definition);
        System.out.println("Serialized XML:\n" + xml);

        // Deserialize the XML back to a Definition object
        Definition deserializedDefinition = xmlMapper.readValue(xml, Definition.class);

        // Verify that the deserialized object matches the original
        assertEquals(1, deserializedDefinition.getRecords().size());
        assertEquals(0, deserializedDefinition.getRecords().get(0).getFields().size());
    }
