@Test
public void testExactOptionNameMatch() throws ParseException {
        // -----------------------------------------------------------------
        // 1️⃣  Build the option set exactly as described in the issue.
        // -----------------------------------------------------------------
        Options options = new Options();

        //  --importToOpen <FILE>
        options.addOption(
                Option.builder()
                        .longOpt("importToOpen")
                        .hasArg()
                        .argName("FILE")
                        .desc("Import to open")
                        .build());

        //  -i, --import <FILE>
        options.addOption(
                Option.builder("i")
                        .longOpt("import")
                        .hasArg()
                        .argName("FILE")
                        .desc("Import")
                        .build());

        // -----------------------------------------------------------------
        // 2️⃣  Parse a command line that uses the *exact* long option name.
        // -----------------------------------------------------------------
        String[] args = {"--import=sample.txt"};

        CommandLineParser parser = new DefaultParser();
        CommandLine cmd;

        try {
            cmd = parser.parse(options, args);
        } catch (AmbiguousOptionException aoe) {
            // The bug manifests itself here – fail the test with a clear message.
            fail("AmbiguousOptionException was thrown for an exact long‑option match: "
                    + aoe.getMessage());
            return; // unreachable, but keeps the compiler happy
        }

        // -----------------------------------------------------------------
        // 3️⃣  Verify that the correct option was selected and its value stored.
        // -----------------------------------------------------------------
        assertTrue("Option \"import\" should be present", cmd.hasOption("import"));
        assertFalse("Option \"importToOpen\" must NOT be present", cmd.hasOption("importToOpen"));

        String value = cmd.getOptionValue("import");
        assertNotNull("Option value for \"import\" must not be null", value);
        assertEquals("sample.txt", value);
    }
