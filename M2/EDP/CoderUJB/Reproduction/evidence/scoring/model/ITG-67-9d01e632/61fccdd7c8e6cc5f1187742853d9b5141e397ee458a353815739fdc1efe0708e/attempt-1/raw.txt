    @Test
    public void testExactOptionNameMatch() throws ParseException {
        Options options = new Options();
        options.addOption(Option.builder().longOpt("importToOpen").hasArg().argName("FILE").build());
        options.addOption(Option.builder("i").longOpt("import").hasArg().argName("FILE").build());

        CommandLine commandLine = new DefaultParser().parse(options, new String[] {"--import=FILE"});

        assertTrue(commandLine.hasOption("import"));
        assertTrue(commandLine.hasOption("i"));
        assertEquals("FILE", commandLine.getOptionValue("import"));
        assertEquals("FILE", commandLine.getOptionValue("i"));
        assertFalse(commandLine.hasOption("importToOpen"));
        assertEquals(0, commandLine.getArgs().length);
    }
