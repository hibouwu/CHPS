@Test
public void testCompressionMethod() throws Exception {
    // Create a new ZipArchiveEntry instance
    ZipArchiveEntry entry = new ZipArchiveEntry("testfile.txt");

    // Define some common compression methods
    int STORED_METHOD = java.util.zip.ZipEntry.STORED;
    int DEFLATED_METHOD = java.util.zip.ZipEntry.DEFLATED;

    // Set the compression method to STORED
    entry.setMethod(STORED_METHOD);

    // Verify that the compression method is set to STORED
    assertEquals("Compression method should be STORED", STORED_METHOD, entry.getMethod());

    // Set the compression method to DEFLATED
    entry.setMethod(DEFLATED_METHOD);

    // Verify that the compression method is set to DEFLATED
    assertEquals("Compression method should be DEFLATED", DEFLATED_METHOD, entry.getMethod());

    // Attempt to set an invalid compression method
    try {
        entry.setMethod(-1); // Invalid compression method
        fail("Expected IllegalArgumentException for invalid compression method");
    } catch (IllegalArgumentException e) {
        // Expected exception
    }

    // Attempt to set another invalid compression method
    try {
        entry.setMethod(9); // Another invalid compression method
        fail("Expected IllegalArgumentException for invalid compression method");
    } catch (IllegalArgumentException e) {
        // Expected exception
    }

    // Set back to a valid method to ensure no side effects from invalid methods
    entry.setMethod(STORED_METHOD);
    assertEquals("Compression method should be STORED after setting a valid method", STORED_METHOD, entry.getMethod());
}
