@Test
public void testDecodeObjectException() {
    // The BinaryCodec instance is created in setUp().
    // Pass an object that is *not* a byte[] or String – the contract
    // of BinaryCodec.decode(Object) states that such a value must cause
    // a DecoderException.
    try {
        instance.decode(new Object());          // illegal argument
        fail("DecoderException expected when decoding an unsupported object type");
    } catch (DecoderException e) {
        // Expected path – the exception message can be inspected if desired
        assertEquals("Input object is not a byte[] or String", e.getMessage());
    }

    // Additionally verify that a null argument also triggers the exception.
    try {
        instance.decode(null);
        fail("DecoderException expected when decoding a null reference");
    } catch (DecoderException e) {
        // Expected – the codec should not accept null as a valid input.
        assertEquals("Input object is null", e.getMessage());
    }
}
