public void testForID_String_old() {
    DateTimeZone zone = DateTimeZone.forID("MET");

    // MET is Middle European Time, not Asia/Tehran.
    long winter = new DateTime(2010, 1, 15, 12, 0, DateTimeZone.UTC).getMillis();
    long summer = new DateTime(2010, 7, 15, 12, 0, DateTimeZone.UTC).getMillis();

    assertEquals("MET must have a standard offset of UTC+01:00",
            60 * 60 * 1000, zone.getStandardOffset(winter));
    assertEquals("MET must use UTC+01:00 in winter",
            60 * 60 * 1000, zone.getOffset(winter));
    assertEquals("MET must use UTC+02:00 during European daylight saving time",
            2 * 60 * 60 * 1000, zone.getOffset(summer));
}
