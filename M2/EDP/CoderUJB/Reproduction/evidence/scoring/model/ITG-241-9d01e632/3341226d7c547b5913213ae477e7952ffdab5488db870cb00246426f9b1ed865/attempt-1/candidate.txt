public void testForID_String_old() {
        // MET should map to Central European Time (CET) rather than Asia/Tehran
        DateTimeZone zone = DateTimeZone.forID("MET");
        assertNotNull("DateTimeZone for 'MET' should not be null", zone);
        // Ensure it is not incorrectly mapped to Asia/Tehran
        assertNotEquals("Asia/Tehran", zone.getID());
        // Verify the standard offset is +01:00 (CET) at a non‑DST date
        DateTime testDate = new DateTime(2020, 1, 1, 0, 0, 0, zone);
        int offsetMillis = zone.getOffset(testDate);
        assertEquals("MET should have an offset of +01:00 (3600000 ms) at standard time",
                     3600000, offsetMillis);
    }
