public void testForID_String_old() {
        // ------------------------------------------------------------
        // 1️⃣  Resolve the legacy ID
        // ------------------------------------------------------------
        DateTimeZone metZone = DateTimeZone.forID("MET");
        assertNotNull("MET should resolve to a DateTimeZone instance", metZone);

        // ------------------------------------------------------------
        // 2️⃣  Guard against the historic bug (wrong mapping to Tehran)
        // ------------------------------------------------------------
        assertNotEquals(
                "MET must NOT be mapped to Asia/Tehran (bug Time‑112)",
                "Asia/Tehran",
                metZone.getID()
        );

        // ------------------------------------------------------------
        // 3️⃣  Verify the correct offsets (Middle European Time)
        // ------------------------------------------------------------
        // a) Winter – standard time (UTC+01:00)
        long winterInstant = new DateTime(2020, 1, 15, 0, 0, DateTimeZone.UTC).getMillis();
        assertEquals(
                "Winter offset for MET should be +01:00",
                3_600_000,                     // 1 hour in milliseconds
                metZone.getOffset(winterInstant)
        );

        // b) Summer – daylight saving time (UTC+02:00)
        long summerInstant = new DateTime(2020, 7, 15, 0, 0, DateTimeZone.UTC).getMillis();
        assertEquals(
                "Summer offset for MET should be +02:00",
                7_200_000,                     // 2 hours in milliseconds
                metZone.getOffset(summerInstant)
        );
    }
