/**
 * A method that returns the maximum value in an ArrayList of integers.
 */
public static Integer max(ArrayList<Integer> arr) {
    Integer max = arr.get(0);
    for (Integer x : arr) {
        if (x > max) {
            max = x;
        }
    }
    return max;
}
