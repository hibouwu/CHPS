@Test
public void testCaverphoneRevisitedCommonCodeAT1111() throws EncoderException {
        final Caverphone1 encoder = createStringEncoder();
        final String[] words = {
            "add", "aid", "at", "art", "eat", "earth", "head", "hit",
            "hot", "hold", "hard", "heart", "it", "out", "old"
        };
        for (final String word : words) {
            Assert.assertEquals("Unexpected encoding for " + word, "AT1111", encoder.encode(word));
        }
    }
