public static synchronized PeriodType forFields(DurationFieldType[] types) {
        if (types == null) {
            throw new IllegalArgumentException("The types array must not be null");
        }
        // Remove duplicates and validate that each type is a standard field type
        List<DurationFieldType> unique = new ArrayList<DurationFieldType>(types.length);
        for (DurationFieldType type : types) {
            if (type == null) {
                throw new IllegalArgumentException("The types array must not contain null");
            }
            // check that the type is one of the eight standard types
            boolean standard = type == DurationFieldType.years()
                    || type == DurationFieldType.months()
                    || type == DurationFieldType.weeks()
                    || type == DurationFieldType.days()
                    || type == DurationFieldType.hours()
                    || type == DurationFieldType.minutes()
                    || type == DurationFieldType.seconds()
                    || type == DurationFieldType.millis();
            if (!standard) {
                throw new IllegalArgumentException("Period type does not support field type: " + type);
            }
            if (!unique.contains(type)) {
                unique.add(type);
            }
        }
        // Order the fields according to the standard order
        DurationFieldType[] standardOrder = new DurationFieldType[] {
                DurationFieldType.years(),
                DurationFieldType.months(),
                DurationFieldType.weeks(),
                DurationFieldType.days(),
                DurationFieldType.hours(),
                DurationFieldType.minutes(),
                DurationFieldType.seconds(),
                DurationFieldType.millis()
        };
        List<DurationFieldType> ordered = new ArrayList<DurationFieldType>(unique.size());
        for (DurationFieldType std : standardOrder) {
            if (unique.contains(std)) {
                ordered.add(std);
            }
        }
        DurationFieldType[] resultTypes = ordered.toArray(new DurationFieldType[ordered.size()]);
        // Build the indices array that maps each field to its standard position
        int[] indices = new int[resultTypes.length];
        for (int i = 0; i < resultTypes.length; i++) {
            DurationFieldType t = resultTypes[i];
            if (t == DurationFieldType.years()) {
                indices[i] = YEAR_INDEX;
            } else if (t == DurationFieldType.months()) {
                indices[i] = MONTH_INDEX;
            } else if (t == DurationFieldType.weeks()) {
                indices[i] = WEEK_INDEX;
            } else if (t == DurationFieldType.days()) {
                indices[i] = DAY_INDEX;
            } else if (t == DurationFieldType.hours()) {
                indices[i] = HOUR_INDEX;
            } else if (t == DurationFieldType.minutes()) {
                indices[i] = MINUTE_INDEX;
            } else if (t == DurationFieldType.seconds()) {
                indices[i] = SECOND_INDEX;
            } else if (t == DurationFieldType.millis()) {
                indices[i] = MILLI_INDEX;
            } else {
                // This should never happen because of the earlier validation
                throw new IllegalArgumentException("Unsupported field type: " + t);
            }
        }
        // Build a readable name for the new PeriodType
        StringBuilder sb = new StringBuilder("PeriodType[");
        for (int i = 0; i < resultTypes.length; i++) {
            if (i > 0) {
                sb.append(',');
            }
            sb.append(resultTypes[i].getName());
        }
        sb.append(']');
        String name = sb.toString();
        PeriodType pt = new PeriodType(name, resultTypes, indices);
        cTypes.put(pt, pt);
        return pt;
    }
