public static synchronized PeriodType forFields(DurationFieldType[] types) {
    // Validate input
    if (types == null) {
        throw new IllegalArgumentException("Types array must not be null");
    }
    
    // Sort the types array to ensure consistent ordering
    Arrays.sort(types, (a, b) -> Integer.compare(a.getType().ordinal(), b.getType().ordinal()));
    
    // Create a key for the map using the sorted types array
    List<DurationFieldType> typeList = Arrays.asList(types);
    PeriodType key = new PeriodType("Custom", types, calculateIndices(types));
    
    // Check if the period type already exists in the cache
    PeriodType existingType = (PeriodType) cTypes.get(key);
    if (existingType != null) {
        return existingType;
    }
    
    // If not, create a new period type and add it to the cache
    PeriodType newType = new PeriodType("Custom", types, calculateIndices(types));
    cTypes.put(newType, newType);
    return newType;
}
