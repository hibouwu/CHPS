    public static byte[] mergeCentralDirectoryData(final ZipExtraField[] data) {
        if (data == null) {
            return null;
        }
        final List<byte[]> result = new ArrayList<>(data.length);
        for (final ZipExtraField ze : data) {
            final byte[] b = ze.getCentralDirectoryData();
            if (b != null) {
                result.add(b);
            }
        }
        return ZipExtraFieldUtils.mergeArrays(result);
    }
