public static byte[] mergeCentralDirectoryData(final ZipExtraField[] data) {
        List<byte[]> dataList = new ArrayList<>();
        int totalLength = 0;

        for (ZipExtraField field : data) {
            byte[] fieldData = field.getCentralDirectoryData();
            dataList.add(fieldData);
            totalLength += fieldData.length;
        }

        byte[] result = new byte[totalLength];
        int offset = 0;
        for (byte[] fieldData : dataList) {
            System.arraycopy(fieldData, 0, result, offset, fieldData.length);
            offset += fieldData.length;
        }

        return result;
    }
