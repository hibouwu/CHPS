    public static byte[] mergeCentralDirectoryData(final ZipExtraField[] data) {
        final boolean lastIsUnparseable = data.length > 0
                && data[data.length - 1] instanceof UnparseableExtraFieldData;
        final int regularCount = lastIsUnparseable ? data.length - 1 : data.length;
        int length = WORD * regularCount;
        for (final ZipExtraField field : data) {
            length += field.getCentralDirectoryLength().getValue();
        }

        final byte[] result = new byte[length];
        int offset = 0;
        for (int i = 0; i < regularCount; i++) {
            final ZipExtraField field = data[i];
            System.arraycopy(field.getHeaderId().getBytes(), 0, result, offset, 2);
            System.arraycopy(field.getCentralDirectoryLength().getBytes(), 0,
                    result, offset + 2, 2);
            offset += WORD;
            final byte[] fieldData = field.getCentralDirectoryData();
            System.arraycopy(fieldData, 0, result, offset, fieldData.length);
            offset += fieldData.length;
        }

        if (lastIsUnparseable) {
            final byte[] fieldData = data[data.length - 1].getCentralDirectoryData();
            System.arraycopy(fieldData, 0, result, offset, fieldData.length);
        }
        return result;
    }
