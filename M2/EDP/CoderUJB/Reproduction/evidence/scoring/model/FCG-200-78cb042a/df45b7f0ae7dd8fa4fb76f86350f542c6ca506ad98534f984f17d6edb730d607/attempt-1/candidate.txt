public static byte[] mergeCentralDirectoryData(final ZipExtraField[] data) {
    final boolean lastIsUnparseableHolder = data.length > 0
            && data[data.length - 1] instanceof UnparseableExtraFieldData;
    final int regularExtraFieldCount = lastIsUnparseableHolder
            ? data.length - 1 : data.length;

    int totalLength = WORD * regularExtraFieldCount;
    for (final ZipExtraField field : data) {
        totalLength += field.getCentralDirectoryLength().getValue();
    }

    final byte[] result = new byte[totalLength];
    int offset = 0;

    for (int i = 0; i < regularExtraFieldCount; i++) {
        final ZipExtraField field = data[i];

        System.arraycopy(field.getHeaderId().getBytes(), 0, result, offset, 2);
        System.arraycopy(field.getCentralDirectoryLength().getBytes(),
                0, result, offset + 2, 2);
        offset += WORD;

        final byte[] fieldData = field.getCentralDirectoryData();
        System.arraycopy(fieldData, 0, result, offset, fieldData.length);
        offset += fieldData.length;
    }

    // Unparseable trailing data has no header or length prefix.
    if (lastIsUnparseableHolder) {
        final byte[] fieldData =
                data[data.length - 1].getCentralDirectoryData();
        System.arraycopy(fieldData, 0, result, offset, fieldData.length);
    }

    return result;
}
