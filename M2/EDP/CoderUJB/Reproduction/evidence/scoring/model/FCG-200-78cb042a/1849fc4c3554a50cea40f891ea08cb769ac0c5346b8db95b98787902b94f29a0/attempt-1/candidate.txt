public static byte[] mergeCentralDirectoryData(final ZipExtraField[] data) {
        if (data == null || data.length == 0) {
            return new byte[0];
        }
        int totalLength = 0;
        for (final ZipExtraField ze : data) {
            if (ze == null) {
                continue;
            }
            final byte[] cd = ze.getCentralDirectoryData();
            totalLength += WORD + (cd != null ? cd.length : 0);
        }
        final byte[] result = new byte[totalLength];
        int offset = 0;
        for (final ZipExtraField ze : data) {
            if (ze == null) {
                continue;
            }
            final ZipShort headerId = ze.getHeaderId();
            final byte[] cd = ze.getCentralDirectoryData();
            final int length = cd != null ? cd.length : 0;
            // header id (2 bytes)
            System.arraycopy(headerId.getBytes(), 0, result, offset, 2);
            offset += 2;
            // data size (2 bytes)
            final ZipShort size = new ZipShort(length);
            System.arraycopy(size.getBytes(), 0, result, offset, 2);
            offset += 2;
            // data bytes
            if (length > 0) {
                System.arraycopy(cd, 0, result, offset, length);
                offset += length;
            }
        }
        return result;
    }
