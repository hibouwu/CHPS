/**
 * A method that finds the maximum value in an unsorted array.
 */
public class MAX {
    public static Integer max(ArrayList<Integer> arr) {
        if (arr.size() == 0) {
            return null;
        } else if (arr.size() == 1) {
            return arr.get(0);
        } else {
            Integer max = arr.get(0);
            for (Integer x : arr) {
                if (x > max) {
                    max = x;
                }
            }
            return max;
        }
    }
}
