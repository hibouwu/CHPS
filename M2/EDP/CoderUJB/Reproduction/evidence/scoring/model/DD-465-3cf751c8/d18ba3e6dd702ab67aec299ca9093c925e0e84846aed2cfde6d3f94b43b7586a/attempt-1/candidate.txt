/**
 * A method that returns the number of occurrences of a specified character in a string.
 */
public class CHARCOUNT {
    public static int charcount(String str, char c) {
        int count = 0;
        for (int i = 0; i < str.length(); i++) {
            if (str.charAt(i) == c) {
                count++;
            }
        }
        return count;
    }
}
