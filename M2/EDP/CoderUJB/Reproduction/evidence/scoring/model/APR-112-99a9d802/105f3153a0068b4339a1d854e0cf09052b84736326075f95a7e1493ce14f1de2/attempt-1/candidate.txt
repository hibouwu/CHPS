public static boolean equal(GeneralPath p1, GeneralPath p2) {
    // ----- null handling -------------------------------------------------
    if (p1 == null) {
        return (p2 == null);
    }
    if (p2 == null) {
        return false;
    }

    // ----- winding rule --------------------------------------------------
    if (p1.getWindingRule() != p2.getWindingRule()) {
        return false;
    }

    // ----- iterate over both paths ---------------------------------------
    PathIterator iterator1 = p1.getPathIterator(null);
    PathIterator iterator2 = p2.getPathIterator(null);   // <-- bug fixed

    double[] d1 = new double[6];
    double[] d2 = new double[6];

    while (!iterator1.isDone() && !iterator2.isDone()) {

        int seg1 = iterator1.currentSegment(d1);
        int seg2 = iterator2.currentSegment(d2);

        // segment type must be identical
        if (seg1 != seg2) {
            return false;
        }

        // compare only the coordinates that are actually used for this segment
        int coordCount;
        switch (seg1) {
            case PathIterator.SEG_MOVETO:
            case PathIterator.SEG_LINETO:
                coordCount = 2;   // x, y
                break;
            case PathIterator.SEG_QUADTO:
                coordCount = 4;   // x1, y1, x2, y2
                break;
            case PathIterator.SEG_CUBICTO:
                coordCount = 6;   // x1, y1, x2, y2, x3, y3
                break;
            case PathIterator.SEG_CLOSE:
                coordCount = 0;   // no coordinates
                break;
            default:
                // Should never happen, but be defensive
                throw new IllegalStateException("Unknown segment type: " + seg1);
        }

        // compare the relevant slice of the coordinate arrays
        for (int i = 0; i < coordCount; i++) {
            if (Double.doubleToLongBits(d1[i]) != Double.doubleToLongBits(d2[i])) {
                return false;
            }
        }

        iterator1.next();
        iterator2.next();
    }

    // Both iterators must finish at the same time
    return iterator1.isDone() && iterator2.isDone();
}
