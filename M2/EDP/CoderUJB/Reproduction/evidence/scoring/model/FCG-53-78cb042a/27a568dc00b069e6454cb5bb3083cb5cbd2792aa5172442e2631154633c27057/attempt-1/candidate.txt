private Node transformAsString(AstNode n) {
    // First let the regular transformer do its job.
    Node result = transform(n);

    // If the result is a NAME token we have to turn it into a STRING token.
    // The identifier of a NAME node is stored in its string field.
    if (result.getToken() == Token.NAME) {
        // Grab the identifier and create a new STRING node.
        String identifier = result.getString();
        result = newStringNode(identifier);
    }

    // If the result is already a STRING token we leave it untouched.
    // (The string literal is already quoted in the source.)

    // Keep the original source‑position information.
    setSourceInfo(result, n);
    return result;
}
