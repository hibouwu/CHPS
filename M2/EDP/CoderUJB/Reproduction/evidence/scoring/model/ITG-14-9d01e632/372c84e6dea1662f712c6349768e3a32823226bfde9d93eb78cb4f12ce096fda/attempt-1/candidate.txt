public void testSymbolTableExpansionBytes() throws Exception {
    final com.fasterxml.jackson.core.JsonFactory factory =
            new com.fasterxml.jackson.core.JsonFactory();
    final java.util.Random random = new java.util.Random(216L);

    // Reuse the factory so parsers also exercise the shared symbol table.
    for (int round = 0; round < 30; ++round) {
        final int fieldCount = 400;
        final String[] names = new String[fieldCount];
        final java.io.ByteArrayOutputStream output =
                new java.io.ByteArrayOutputStream();

        try (com.fasterxml.jackson.core.JsonGenerator generator =
                     factory.createGenerator(output)) {
            generator.writeStartObject();
            for (int i = 0; i < fieldCount; ++i) {
                // Names longer than 12 UTF-8 bytes exercise long-name storage
                // while unique names force symbol-table expansion.
                names[i] = "long_symbol_table_field_" + round + "_" + i + "_"
                        + Long.toHexString(random.nextLong());
                generator.writeNumberField(names[i], i);
            }
            generator.writeEndObject();
        }

        // A byte-array parser uses ByteQuadsCanonicalizer, where the
        // reported ArrayIndexOutOfBoundsException occurred.
        try (com.fasterxml.jackson.core.JsonParser parser =
                     factory.createParser(output.toByteArray())) {
            assertEquals(com.fasterxml.jackson.core.JsonToken.START_OBJECT,
                    parser.nextToken());

            for (int i = 0; i < fieldCount; ++i) {
                assertEquals(names[i], parser.nextFieldName());
                assertEquals(com.fasterxml.jackson.core.JsonToken.VALUE_NUMBER_INT,
                        parser.nextToken());
                assertEquals(i, parser.getIntValue());
            }

            assertEquals(com.fasterxml.jackson.core.JsonToken.END_OBJECT,
                    parser.nextToken());
            assertNull(parser.nextToken());
        }
    }
}
