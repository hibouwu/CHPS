public void testSymbolTableExpansionBytes() throws Exception {
        com.fasterxml.jackson.core.JsonFactory factory =
                new com.fasterxml.jackson.core.JsonFactory();
        java.util.Random random = new java.util.Random(216L);

        for (int round = 0; round < 12; ++round) {
            String[] names = new String[2000];
            java.io.ByteArrayOutputStream output =
                    new java.io.ByteArrayOutputStream();
            com.fasterxml.jackson.core.JsonGenerator generator =
                    factory.createGenerator(output);

            try {
                generator.writeStartObject();
                for (int i = 0; i < names.length; ++i) {
                    StringBuilder name = new StringBuilder();
                    name.append("field_").append(round).append('_').append(i);
                    for (int j = 0; j < 8; ++j) {
                        name.append('_').append(Long.toHexString(random.nextLong()));
                    }
                    names[i] = name.toString();
                    generator.writeNumberField(names[i], i);
                }
                generator.writeEndObject();
            } finally {
                generator.close();
            }

            com.fasterxml.jackson.core.JsonParser parser =
                    factory.createParser(output.toByteArray());
            try {
                assertEquals(com.fasterxml.jackson.core.JsonToken.START_OBJECT,
                        parser.nextToken());
                for (int i = 0; i < names.length; ++i) {
                    assertEquals(names[i], parser.nextFieldName());
                    assertEquals(names[i], parser.getCurrentName());
                    assertEquals(com.fasterxml.jackson.core.JsonToken.VALUE_NUMBER_INT,
                            parser.nextToken());
                    assertEquals(i, parser.getIntValue());
                }
                assertEquals(com.fasterxml.jackson.core.JsonToken.END_OBJECT,
                        parser.nextToken());
                assertNull(parser.nextToken());
            } finally {
                parser.close();
            }
        }
    }
