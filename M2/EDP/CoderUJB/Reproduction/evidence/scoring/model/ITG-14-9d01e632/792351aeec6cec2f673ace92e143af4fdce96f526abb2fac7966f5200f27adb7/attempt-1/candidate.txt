public void testSymbolTableExpansionBytes() throws Exception {
        ObjectMapper objectMapper = new ObjectMapper();
        int numberOfEntries = 1000; // Adjust this number to increase the load on the symbol table

        // Create a large map to serialize and deserialize
        Map<String, String> largeMap = new HashMap<>();
        for (int i = 0; i < numberOfEntries; i++) {
            largeMap.put("key" + i, "value" + i);
        }

        // Serialize the map to a byte array
        byte[] serializedBytes = objectMapper.writeValueAsBytes(largeMap);

        // Deserialize the byte array back to a map multiple times
        for (int i = 0; i < 100; i++) { // Repeat the deserialization process to increase the chance of triggering the issue
            Map<?, ?> deserializedMap = objectMapper.readValue(serializedBytes, Map.class);
            if (deserializedMap.size() != numberOfEntries) {
                throw new AssertionError("Deserialized map size does not match the original map size");
            }
        }
    }
