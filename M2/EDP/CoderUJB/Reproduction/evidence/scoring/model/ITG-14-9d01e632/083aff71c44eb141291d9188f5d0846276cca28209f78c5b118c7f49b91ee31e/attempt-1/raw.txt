    public void testSymbolTableExpansionBytes() throws Exception {
        ObjectMapper objectMapper = new ObjectMapper();
        StringBuilder longKeyBuilder = new StringBuilder();
        for (int i = 0; i < 128; i++) {
            longKeyBuilder.append("a");
        }
        String longKey = longKeyBuilder.toString();
        Map<String, Object> testData = new HashMap<>();
        testData.put(longKey, "value");

        for (int i = 0; i < 1000; i++) {
            byte[] bytes = objectMapper.writeValueAsBytes(testData);
            Map<?, ?> result = objectMapper.readValue(bytes, Map.class);
            assertEquals("value", result.get(longKey));
        }
    }
