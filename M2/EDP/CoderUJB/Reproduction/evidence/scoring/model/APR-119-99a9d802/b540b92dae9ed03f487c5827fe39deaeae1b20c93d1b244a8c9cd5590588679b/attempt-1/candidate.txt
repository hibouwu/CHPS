protected void _serializeXmlNull(JsonGenerator jgen) throws IOException
{
    // 14-Nov-2016, tatu: As per [dataformat-xml#213], we may have explicitly
    //    configured root name...
    if (jgen instanceof ToXmlGenerator) {
        ToXmlGenerator xgen = (ToXmlGenerator) jgen;
        // Check if we should write a root element for null
        if (xgen.getOutputContext().inRoot()) {
            // Optionally, you can choose to write an empty element or an attribute
            // For now, we choose to do nothing (omit the element)
            return;
        }
    }
    // If not in root context, or not a ToXmlGenerator, fall back to default behavior
    super.serializeValue(jgen, null);
}
