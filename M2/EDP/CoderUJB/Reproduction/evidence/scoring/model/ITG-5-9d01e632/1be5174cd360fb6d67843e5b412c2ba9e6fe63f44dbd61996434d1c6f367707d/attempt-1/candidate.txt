@Test
public void testReadingOfFirstStoredEntry() throws Exception {
        try (ByteArrayInputStream bais = new ByteArrayInputStream(ZIP_DATA);
             ZipArchiveInputStream zais = new ZipArchiveInputStream(bais)) {

            ZipArchiveEntry entry = zais.getNextZipEntry();
            assertNotNull(entry, "The ZIP entry should not be null");
            assertEquals("test.txt", entry.getName(), "The entry name should be 'test.txt'");

            StringBuilder contentBuilder = new StringBuilder();
            int len;
            byte[] buffer = new byte[1024];
            while ((len = zais.read(buffer)) > 0) {
                contentBuilder.append(new String(buffer, 0, len, StandardCharsets.UTF_8));
            }

            String content = contentBuilder.toString().trim();
            assertEquals("data", content, "The content of test.txt should be 'data'");
        }
    }
