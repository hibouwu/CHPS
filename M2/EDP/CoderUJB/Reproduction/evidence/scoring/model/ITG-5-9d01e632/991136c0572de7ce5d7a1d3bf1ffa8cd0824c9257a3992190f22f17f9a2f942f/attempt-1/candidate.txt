@Test
public void testReadingOfFirstStoredEntry() throws Exception {
    // Local file header and STORED entry data from COMPRESS-264.
    final byte[] zip = {
        0x50, 0x4b, 0x03, 0x04, 0x0a, 0x00,
        0x00, 0x00, 0x00, 0x00, 0x03, 0x7b,
        (byte) 0xd1, 0x42, (byte) 0x82, (byte) 0xc5,
        (byte) 0xc1, (byte) 0xe6,
        0x05, 0x00, 0x00, 0x00, 0x05, 0x00,
        0x00, 0x00, 0x08, 0x00, 0x1c, 0x00,
        0x74, 0x65, 0x73, 0x74, 0x2e, 0x74,
        0x78, 0x74, 0x55, 0x54, 0x09, 0x00,
        0x03, 0x56, 0x62, (byte) 0xbf, 0x51, 0x2a,
        0x63, (byte) 0xbf, 0x51, 0x75, 0x78, 0x0b,
        0x00, 0x01, 0x04, 0x01, (byte) 0xff, 0x01,
        0x00, 0x04, (byte) 0x88, 0x13, 0x00, 0x00,
        0x64, 0x61, 0x74, 0x61, 0x0a
    };
    final byte[] expected = { 0x64, 0x61, 0x74, 0x61, 0x0a };

    // Verify both bulk reads and single-byte reads of the first entry.
    for (int mode = 0; mode < 2; mode++) {
        final ZipArchiveInputStream in = new ZipArchiveInputStream(
                new java.io.ByteArrayInputStream(zip));
        try {
            final ZipArchiveEntry entry = in.getNextZipEntry();
            org.junit.Assert.assertNotNull(entry);
            org.junit.Assert.assertEquals("test.txt", entry.getName());
            org.junit.Assert.assertEquals(
                    java.util.zip.ZipEntry.STORED, entry.getMethod());
            org.junit.Assert.assertEquals(5L, entry.getSize());

            final byte[] actual = new byte[expected.length];
            if (mode == 0) {
                int offset = 0;
                while (offset < actual.length) {
                    final int count = in.read(
                            actual, offset, actual.length - offset);
                    org.junit.Assert.assertTrue(
                            "Expected entry data rather than EOF or an empty read",
                            count > 0);
                    offset += count;
                }
            } else {
                for (int i = 0; i < actual.length; i++) {
                    final int value = in.read();
                    org.junit.Assert.assertTrue(
                            "Unexpected end of entry", value >= 0);
                    actual[i] = (byte) value;
                }
            }

            org.junit.Assert.assertArrayEquals(expected, actual);
            org.junit.Assert.assertEquals(-1, in.read());
        } finally {
            in.close();
        }
    }
}
