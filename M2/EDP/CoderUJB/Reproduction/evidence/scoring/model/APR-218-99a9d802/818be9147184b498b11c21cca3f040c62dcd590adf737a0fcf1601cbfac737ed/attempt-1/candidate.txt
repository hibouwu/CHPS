private boolean compute(Object left, Object right) {
    // 1️⃣ Reduce both operands (unwrap collections, evaluate node‑sets, …)
    left = reduce(left);
    right = reduce(right);

    // 2️⃣ Reset any InitialContext so that iteration starts from the beginning
    if (left instanceof InitialContext) {
        ((InitialContext) left).reset();
    }
    if (right instanceof InitialContext) {
        ((InitialContext) right).reset();
    }

    // 3️⃣ Iterator‑based comparisons (node‑sets, collections, etc.)
    if (left instanceof Iterator && right instanceof Iterator) {
        return findMatch((Iterator) left, (Iterator) right);
    }
    if (left instanceof Iterator) {
        return containsMatch((Iterator) left, right);
    }
    if (right instanceof Iterator) {
        return containsMatch((Iterator) right, left);
    }

    // 4️⃣ Type‑aware scalar comparison
    //    InfoSetUtil.compareObjects returns:
    //      <0  if left  < right
    //       0  if left == right
    //      >0  if left  > right
    int compare = InfoSetUtil.compareObjects(left, right);

    // 5️⃣ Let the concrete subclass decide what the comparison result means
    return evaluateCompare(compare);
}
