private boolean compute(Object left, Object right) {
    left = reduce(left);
    right = reduce(right);

    // Handle InitialContext reset
    if (left instanceof InitialContext) {
        ((InitialContext) left).reset();
    }
    if (right instanceof InitialContext) {
        ((InitialContext) right).reset();
    }

    // Handle Iterator cases
    if (left instanceof Iterator<?> && right instanceof Iterator<?>) {
        return findMatch((Iterator<?>) left, (Iterator<?>) right);
    }
    if (left instanceof Iterator<?>) {
        return containsMatch((Iterator<?>) left, right);
    }
    if (right instanceof Iterator<?>) {
        return containsMatch((Iterator<?>) right, left);
    }

    // Handle null values
    if (left == null || right == null) {
        return evaluateCompare(left == right ? 0 : -1); // Assuming null == null is true
    }

    // Attempt to compare as numbers
    try {
        double ld = InfoSetUtil.doubleValue(left);
        double rd = InfoSetUtil.doubleValue(right);
        return evaluateCompare(Double.compare(ld, rd));
    } catch (NumberFormatException e) {
        // Fallback to string comparison if conversion to double fails
        String ls = InfoSetUtil.stringValue(left);
        String rs = InfoSetUtil.stringValue(right);
        return evaluateCompare(ls.compareTo(rs));
    }
}
