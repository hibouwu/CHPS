public void testCheckSymbolsOverrideForQuiet() throws IOException {
        // Create a simple JavaScript source file with undefined variables and global 'this' usage
        String sourceCode = "foo = {};\nfoo.bar = function() { alert(this.baz); };";
        SourceFile inputSource = SourceFile.fromCode("test.js", sourceCode);

        // Set up the compiler options
        CompilerOptions options = new CompilerOptions();
        options.setWarningLevel(CompilerOptions.WarningLevel.QUIET);
        options.addWarningsGuard(new CommandLineRunner.WarningsGuard.Builder()
                .addGroupWarnings("checkVars", CheckLevel.WARNING)
                .addGroupWarnings("undefinedVars", CheckLevel.ERROR)
                .build());

        // Initialize the compiler and compile the source code
        Compiler compiler = new Compiler();
        Result result = compiler.compile(List.of(), List.of(inputSource), options);

        // Verify that there is one error (for undefined variable) and one warning (for global 'this')
        Assert.assertEquals(1, result.errors.length);
        Assert.assertEquals(1, result.warnings.length);

        // Check the specific error and warning messages
        JSError error = result.errors[0];
        Assert.assertEquals("variable foo is undefined", error.description);

        JSError warning = result.warnings[0];
        Assert.assertEquals("dangerous use of the global this object", warning.description);
    }
