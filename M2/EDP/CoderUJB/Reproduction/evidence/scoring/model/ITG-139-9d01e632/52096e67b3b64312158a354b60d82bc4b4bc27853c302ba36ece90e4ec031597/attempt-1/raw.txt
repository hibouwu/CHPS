public void testCheckSymbolsOverrideForQuiet() {
    String code = "foo = {};\nfoo.bar = function() { alert(this.baz); };";

    // Quiet mode: simulate --jscomp_warning checkVars (should not produce warnings)
    CompilerOptions optsQuiet = new CompilerOptions();
    // Enable the checkVars diagnostic group, but keep default warning level (QUIET)
    optsQuiet.setWarningLevel(DiagnosticGroups.CHECK_VARIABLES, CheckLevel.WARNING);
    // Do not enable undefinedVars explicitly
    Compiler compilerQuiet = new Compiler();
    compilerQuiet.compile(
        Collections.emptyList(),
        Collections.singletonList(SourceFile.fromCode("test.js", code)),
        optsQuiet);
    // Expect no errors or warnings in quiet mode
    assertEquals(0, compilerQuiet.getErrorManager().getErrorCount());
    assertEquals(0, compilerQuiet.getErrorManager().getWarningCount());

    // Verbose mode: simulate --warning_level VERBOSE (should produce undefined variable warning)
    CompilerOptions optsVerbose = new CompilerOptions();
    // Set all diagnostics to VERBOSE to mimic the command line flag
    optsVerbose.setWarningLevel(DiagnosticGroups.ALL_DIAGNOSTICS, CheckLevel.VERBOSE);
    Compiler compilerVerbose = new Compiler();
    compilerVerbose.compile(
        Collections.emptyList(),
        Collections.singletonList(SourceFile.fromCode("test.js", code)),
        optsVerbose);
    // Expect an undefined variable error (variable foo is undefined)
    assertEquals(1, compilerVerbose.getErrorManager().getErrorCount());
    assertTrue(
        compilerVerbose.getErrorManager().getErrors().get(0).getDescription()
            .contains("variable foo is undefined"));
}
