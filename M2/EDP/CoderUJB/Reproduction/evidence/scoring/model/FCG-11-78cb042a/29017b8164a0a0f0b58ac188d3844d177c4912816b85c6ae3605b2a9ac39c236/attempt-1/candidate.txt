@SuppressWarnings("resource")
public byte[] quoteAsUTF8(String text) {
    if (text == null) {
        return new byte[0]; // Return an empty byte array for null input
    }

    // Ensure we have a ByteArrayBuilder to use
    if (_bytes == null) {
        _bytes = BufferRecyclers.getByteBuffer();
    } else {
        _bytes.reset();
    }

    final int len = text.length();
    ByteArrayBuilder bb = _bytes;
    int ptr = 0;

    for (int i = 0; i < len; ++i) {
        char c = text.charAt(i);
        if (c < 0x80) { // ASCII characters
            if (c == '"') {
                bb.appendByte((byte) '\\');
                bb.appendByte((byte) '"');
            } else if (c == '\\') {
                bb.appendByte((byte) '\\');
                bb.appendByte((byte) '\\');
            } else {
                bb.appendByte((byte) c);
            }
        } else if (c < 0x800) { // 2-byte characters
            bb.appendByte((byte) (0xc0 | (c >> 6)));
            bb.appendByte((byte) (0x80 | (c & 0x3f)));
        } else if (c >= SURR1_FIRST && c <= SURR1_LAST) { // High surrogate
            // Must ensure we have a low surrogate next
            if (i + 1 < len) {
                char d = text.charAt(i + 1);
                if (d >= SURR2_FIRST && d <= SURR2_LAST) {
                    int codePoint = _convert(c, d);
                    bb.appendByte((byte) (0xf0 | (codePoint >> 18)));
                    bb.appendByte((byte) (0x80 | ((codePoint >> 12) & 0x3f)));
                    bb.appendByte((byte) (0x80 | ((codePoint >> 6) & 0x3f)));
                    bb.appendByte((byte) (0x80 | (codePoint & 0x3f)));
                    ++i; // Skip the low surrogate
                } else {
                    _illegal(c); // Handle illegal surrogate pair
                }
            } else {
                _illegal(c); // Handle unpaired high surrogate
            }
        } else if (c >= SURR2_FIRST && c <= SURR2_LAST) { // Low surrogate without preceding high surrogate
            _illegal(c);
        } else { // 3-byte characters
            bb.appendByte((byte) (0xe0 | (c >> 12)));
            bb.appendByte((byte) (0x80 | ((c >> 6) & 0x3f)));
            bb.appendByte((byte) (0x80 | (c & 0x3f)));
        }
    }
