@SuppressWarnings("resource")
public byte[] quoteAsUTF8(String text)
{
    ByteArrayBuilder bytes = _bytes;
    if (bytes == null) {
        _bytes = bytes = new ByteArrayBuilder();
    } else {
        bytes.reset();
    }

    final int[] escapeCodes = CharTypes.get7BitOutputEscapes();
    final int length = text.length();

    for (int i = 0; i < length; ++i) {
        int ch = text.charAt(i);

        if (ch <= 0x7F) {
            int escape = escapeCodes[ch];
            if (escape == 0) {
                bytes.append(ch);
            } else {
                bytes.append('\\');
                if (escape > 0) {
                    bytes.append(escape);
                } else {
                    bytes.append('u');
                    bytes.append('0');
                    bytes.append('0');
                    bytes.append(HB[ch >> 4]);
                    bytes.append(HB[ch & 0x0F]);
                }
            }
        } else if (ch <= 0x7FF) {
            bytes.append(0xC0 | (ch >> 6));
            bytes.append(0x80 | (ch & 0x3F));
        } else if (ch < SURR1_FIRST || ch > SURR2_LAST) {
            bytes.append(0xE0 | (ch >> 12));
            bytes.append(0x80 | ((ch >> 6) & 0x3F));
            bytes.append(0x80 | (ch & 0x3F));
        } else {
            // A surrogate must be a high surrogate followed by a low surrogate.
            if (ch > SURR1_LAST) {
                _illegal(ch);
            }
            if (i + 1 >= length) {
                _illegal(ch);
            }

            int codePoint = _convert(ch, text.charAt(++i));
            bytes.append(0xF0 | (codePoint >> 18));
            bytes.append(0x80 | ((codePoint >> 12) & 0x3F));
            bytes.append(0x80 | ((codePoint >> 6) & 0x3F));
            bytes.append(0x80 | (codePoint & 0x3F));
        }
    }

    return bytes.toByteArray();
}
