@SuppressWarnings("resource")
public byte[] quoteAsUTF8(String text) {
        if (text == null) {
            return null;
        }
        // First, quote the string (adds JSON escapes)
        char[] quoted = quoteAsString(text);
        // Reuse or create a ByteArrayBuilder for output
        ByteArrayBuilder bb = _bytes;
        if (bb == null) {
            bb = new ByteArrayBuilder(BufferRecyclers.getByteArrayBuilder());
            _bytes = bb;
        } else {
            bb.reset();
        }
        // Encode the quoted characters as UTF-8
        final int len = quoted.length;
        for (int i = 0; i < len; ++i) {
            int c = quoted[i];
            if (c < 0x80) {
                bb.append(c);
            } else if (c < 0x800) {
                bb.append(0xC0 | (c >> 6));
                bb.append(0x80 | (c & 0x3F));
            } else {
                bb.append(0xE0 | (c >> 12));
                bb.append(0x80 | ((c >> 6) & 0x3F));
                bb.append(0x80 | (c & 0x3F));
            }
        }
        return bb.toByteArray();
    }
