@SuppressWarnings("resource")
public byte[] quoteAsUTF8(String text)
    {
        ByteArrayBuilder bytes = _bytes;
        if (bytes == null) {
            _bytes = bytes = new ByteArrayBuilder(null);
        } else {
            bytes.reset();
        }

        final int[] escapes = CharTypes.get7BitOutputEscapes();
        final int length = text.length();

        for (int i = 0; i < length; ++i) {
            int ch = text.charAt(i);

            if (ch < 0x80) {
                int escape = escapes[ch];
                if (escape == 0) {
                    bytes.append(ch);
                } else {
                    bytes.append('\\');
                    if (escape > 0) {
                        bytes.append(escape);
                    } else {
                        bytes.append('u');
                        bytes.append('0');
                        bytes.append('0');
                        bytes.append(HB[ch >> 4]);
                        bytes.append(HB[ch & 0xF]);
                    }
                }
            } else if (ch < 0x800) {
                bytes.append(0xC0 | (ch >> 6));
                bytes.append(0x80 | (ch & 0x3F));
            } else if (ch < SURR1_FIRST || ch > SURR2_LAST) {
                bytes.append(0xE0 | (ch >> 12));
                bytes.append(0x80 | ((ch >> 6) & 0x3F));
                bytes.append(0x80 | (ch & 0x3F));
            } else {
                if (ch > SURR1_LAST || i + 1 >= length) {
                    _illegal(ch);
                }
                ch = _convert(ch, text.charAt(++i));
                bytes.append(0xF0 | (ch >> 18));
                bytes.append(0x80 | ((ch >> 12) & 0x3F));
                bytes.append(0x80 | ((ch >> 6) & 0x3F));
                bytes.append(0x80 | (ch & 0x3F));
            }
        }

        return bytes.toByteArray();
    }
