public int parseInto(ReadWritableInstant instant, String text, int position) {
    // Ensure a parser is present; otherwise an UnsupportedOperationException
    // is thrown by requireParser().
    DateTimeParser parser = requireParser();

    if (instant == null) {
        throw new IllegalArgumentException("Instant must not be null");
    }

    // The instant's current millis and chronology are used as the default
    // values for any fields that are not present in the text.
    long instantMillis = instant.getMillis();
    Chronology chrono = instant.getChronology();

    // Convert the instant to a *local* instant (millis in the instant's
    // current time‑zone) – this is what the parser bucket expects.
    long instantLocal = instantMillis + chrono.getZone().getOffset(instantMillis);

    // Apply any formatter‑level overrides (zone, chronology, etc.).
    chrono = selectChronology(chrono);
