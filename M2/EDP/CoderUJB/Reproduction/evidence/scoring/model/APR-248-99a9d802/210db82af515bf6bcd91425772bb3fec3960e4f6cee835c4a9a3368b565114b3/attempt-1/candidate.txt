    /**
     * Outputs the record separator.
     *
     * @throws IOException
     *             If an I/O error occurs
     */
    public void println() throws IOException {
        final String recordSeparator = format.getRecordSeparator();
        if (recordSeparator.equals(LF)) {
            out.append(LF);
        } else if (recordSeparator.equals(CR)) {
            out.append(CR);
        } else if (recordSeparator.equals(CRLF)) {
            out.append(CRLF);
        } else if (recordSeparator.equals(LFCR)) {
            out.append(LFCR);
        } else {
            out.append(recordSeparator);
        }
        newRecord = true;
    }
