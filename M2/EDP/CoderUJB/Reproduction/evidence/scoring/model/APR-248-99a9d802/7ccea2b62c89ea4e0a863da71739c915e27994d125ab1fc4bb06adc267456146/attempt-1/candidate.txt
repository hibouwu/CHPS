public void println() throws IOException {
    // The record separator may be user‑defined (e.g. "\n") or {@code null}
    // when the default (CR + LF) should be used.
    final String recordSeparator = format.getRecordSeparator();

    if (recordSeparator != null) {
        // Use the user‑provided separator.
        out.append(recordSeparator);
    } else {
        // No custom separator – fall back to the library default.
        out.append(CR);
        out.append(LF);
    }

    // After a record separator we are ready for a new record.
    newRecord = true;
}
