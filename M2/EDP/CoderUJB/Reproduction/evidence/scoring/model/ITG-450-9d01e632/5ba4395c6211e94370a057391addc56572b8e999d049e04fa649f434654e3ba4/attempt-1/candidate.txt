@Test
public void shouldMatchCaptureArgumentsWhenArgsCountDoesNOTMatch() throws Exception {
        // Arrange
        MyService myService = mock(MyService.class);
        ArgumentCaptor<String> arg1Captor = ArgumentCaptor.forClass(String.class);
        ArgumentCaptor<String> arg2Captor = ArgumentCaptor.forClass(String.class);

        // Act
        myService.performAction("value1", "value2");

        // Assert
        verify(myService).performAction(arg1Captor.capture(), arg2Captor.capture());

        // This assertion will fail because we are trying to capture more arguments than we have
        List<String> capturedArgs1 = arg1Captor.getAllValues();
        List<String> capturedArgs2 = arg2Captor.getAllValues();

        assertEquals(1, capturedArgs1.size());
        assertEquals(1, capturedArgs2.size());

        // Intentionally causing an exception to demonstrate the test case
        verify(myService).performAction(arg1Captor.capture(), arg2Captor.capture(), arg1Captor.capture());
    }
