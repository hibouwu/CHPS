@Test
public void shouldMatchCaptureArgumentsWhenArgsCountDoesNOTMatch() throws Exception {
        java.io.PrintStream mock = org.mockito.Mockito.mock(java.io.PrintStream.class);
        org.mockito.ArgumentCaptor<Object> captor =
                org.mockito.ArgumentCaptor.forClass(Object.class);

        mock.printf("%s %s", "first", "second");

        boolean verificationFailed = false;
        try {
            org.mockito.Mockito.verify(mock).printf(
                    org.mockito.Mockito.eq("%s %s"), captor.capture());
        } catch (AssertionError expected) {
            verificationFailed = true;
        }

        org.junit.Assert.assertTrue(
                "Verification should fail when argument counts differ", verificationFailed);
        org.junit.Assert.assertTrue(
                "Arguments should not be captured from a nonmatching invocation",
                captor.getAllValues().isEmpty());

        org.mockito.Mockito.verify(mock).printf("%s %s", "first", "second");
    }
