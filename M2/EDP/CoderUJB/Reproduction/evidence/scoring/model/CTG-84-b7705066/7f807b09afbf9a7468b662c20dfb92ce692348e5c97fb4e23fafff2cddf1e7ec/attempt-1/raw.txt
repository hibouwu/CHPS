    /**
     * Tests reusing a parser to process new string records one at a time as they are being discovered. See [CSV-110].
     *
     * @throws IOException
     */
    @Test
    public void testGetOneLineOneParser() throws IOException {
        final CSVParser parser = new CSVParserBuilder().build();
        
        // Test with first input
        parser.parse(new StringReader(CSV_INPUT_1));
        Record record = parser.getRecord();
        assertNotNull(record);
        assertArrayEquals(new String[]{"a", "b", "c", "d"}, record.values());

        // Test with second input
        parser.parse(new StringReader(CSV_INPUT_2));
        record = parser.getRecord();
        assertNotNull(record);
        assertArrayEquals(new String[]{"a", "b", "1 2"}, record.values());

        // Ensure no more records are available
        assertNull(parser.getRecord());

        parser.close();
    }
