    @Test
    public void testGetOneLineOneParser() throws IOException {
        final CSVParser parser = new CSVParser(new StringReader(CSV_INPUT_1), CSVFormat.DEFAULT);
        final String[] firstLine = parser.getLine();
        assertNotNull(firstLine);
        assertEquals(4, firstLine.length);
        assertEquals("a", firstLine[0]);
        assertEquals("b", firstLine[1]);
        assertEquals("c", firstLine[2]);
        assertEquals("d", firstLine[3]);
        final String[] secondLine = parser.getLine();
        assertNotNull(secondLine);
        assertEquals(2, secondLine.length);
        assertEquals("a", secondLine[0]);
        assertEquals("b", secondLine[1]);
        assertEquals("1 2", secondLine[2]);
        assertNull(parser.getLine());
        parser.close();
    }
