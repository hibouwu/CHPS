@Test
public void testGetOneLineOneParser() throws IOException {
    try (java.io.PipedReader reader = new java.io.PipedReader();
            java.io.PipedWriter writer = new java.io.PipedWriter(reader);
            CSVParser parser = new CSVParser(reader, CSVFormat.DEFAULT)) {

        final String[] inputs = { CSV_INPUT_1, CSV_INPUT_2 };

        for (int i = 0; i < inputs.length; i++) {
            writer.write(inputs[i]);
            writer.write('\n');
            writer.flush();

            final CSVRecord record = parser.nextRecord();
            assertNotNull(record);
            assertEquals(RESULT[i].length, record.size());
            for (int j = 0; j < RESULT[i].length; j++) {
                assertEquals(RESULT[i][j], record.get(j));
            }
            assertEquals((long) i + 1, record.getRecordNumber());
        }

        writer.close();
        assertNull(parser.nextRecord());
    }
}
