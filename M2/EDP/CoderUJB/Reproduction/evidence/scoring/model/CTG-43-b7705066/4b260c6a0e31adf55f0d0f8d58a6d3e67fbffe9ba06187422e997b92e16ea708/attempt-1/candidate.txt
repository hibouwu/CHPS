public void testMethodsThatShouldAlwaysReturnNull() {
        // -----------------------------------------------------------------
        // Create an UnsupportedDateTimeField instance.  The concrete
        // DurationField we pass (weeks) is irrelevant – the class stores it
        // but never uses it for the two methods we are testing.
        // -----------------------------------------------------------------
        DateTimeField field = UnsupportedDateTimeField.getInstance(
                hourOfDayType,
                weeksDuration);

        // -----------------------------------------------------------------
        // Verify the contract: both methods must return null.
        // -----------------------------------------------------------------
        assertNull(
                "UnsupportedDateTimeField#getRangeDurationField() must return null",
                field.getRangeDurationField());

        assertNull(
                "UnsupportedDateTimeField#getLeapDurationField() must return null",
                field.getLeapDurationField());
    }
