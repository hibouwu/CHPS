public PeriodConverter removePeriodConverter(PeriodConverter converter)
        throws SecurityException {

    // Null arguments are ignored – nothing to remove.
    if (converter == null) {
        return null;
    }

    // Ensure the caller has permission to alter the period converters.
    checkAlterPeriodConverters();

    // The underlying ConverterSet handles the actual removal and
    // returns the previously‑registered converter (or null if it was
    // not present).  Cast the result back to PeriodConverter.
    synchronized (this) {
        return (PeriodConverter) iPeriodConverters.remove(converter);
    }
}
