public Element getElementById(String id) {
        if (id == null)
            return null;
        // check this element first
        if (id.equals(this.id()))
            return this;
        // getAllElements includes this element, so we can iterate and skip the first occurrence
        for (Element el : this.getAllElements()) {
            if (el == this)
                continue;
            if (id.equals(el.id()))
                return el;
        }
        return null;
    }
