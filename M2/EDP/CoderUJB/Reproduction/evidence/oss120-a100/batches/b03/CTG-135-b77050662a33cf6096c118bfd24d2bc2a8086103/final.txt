package org.joda.time;

import java.util.Comparator;
import junit.framework.TestCase;

/**
 * Test cases for {@link DateTimeComparator}.
 *
 * This file contains the implementation of the {@code testSecond} method
 * that verifies the behaviour of a comparator that works at the *second*
 * granularity (i.e. milliseconds are ignored).
 *
 * The test follows the same pattern that the original Joda‑Time test suite
 * uses for the other field‑level comparators (millis, minute, hour, …).
 *
 * @author  Your Name
 */
public class TestDateTimeComparator extends TestCase {

    // -----------------------------------------------------------------------
    // Test fixtures
    // -----------------------------------------------------------------------
    private static final Chronology ISO = ISOChronology.getInstance();

    /** The two DateTime objects that will be compared in the tests. */
    private DateTime aDateTime;
    private DateTime bDateTime;

    /** Comparator that works at the second level. */
    private Comparator<Object> cSecond;

    // -----------------------------------------------------------------------
    // JUnit infrastructure
    // -----------------------------------------------------------------------
    public TestDateTimeComparator(String name) {
        super(name);
    }

    @Override
    protected void setUp() throws Exception {
        super.setUp();

        /*  Initialise the comparator that ignores everything finer than a
            second.  The static factory method that accepts a *lower* limit
            creates a comparator that compares from that field up to the most
            significant field (year).  Passing {@code DateTimeFieldType.secondOfMinute()}
            therefore yields a comparator that treats two instants that differ
            only in milliseconds as equal. */
        cSecond = DateTimeComparator.getInstance(DateTimeFieldType.secondOfMinute());
    }

    // -----------------------------------------------------------------------
    // The required test method
    // -----------------------------------------------------------------------
    /**
     * Test unequal comparisons with second comparators.
     *
     * <p>We verify three things:
     *
     * <ul>
     *   <li>Two instants that fall in the same second compare as equal
     *       (the comparator must ignore the millisecond part).</li>
     *   <li>If the first instant is later than the second (by at least one
     *       whole second) the comparator returns a positive value.</li>
     *   <li>If the first instant is earlier than the second the comparator
     *       returns a negative value.</li>
     * </ul>
     *
     * The test uses fixed timestamps (no {@code System.currentTimeMillis()})
     * to avoid the flakiness that the original Joda‑Time test suite reported.
     */
    public void testSecond() {
        /* --------------------------------------------------------------------
         * 1) Equality when only the millisecond part differs.
         * -------------------------------------------------------------------- */
        aDateTime = new DateTime(2005, 6, 9, 12, 30, 45, 123, DateTimeZone.UTC);
        bDateTime = new DateTime(2005, 6, 9, 12, 30, 45, 999, DateTimeZone.UTC);

        // The two objects are in the same second → comparator must return 0
        assertEquals("Second‑Equality (DateTime)", 0,
                     cSecond.compare(aDateTime, bDateTime));

        // The same check using java.util.Date (the comparator accepts any Object)
        assertEquals("Second‑Equality (java.util.Date)", 0,
                     cSecond.compare(aDateTime.toDate(), bDateTime.toDate()));

        /* --------------------------------------------------------------------
         * 2) Ordering when the seconds differ.
         * -------------------------------------------------------------------- */
        // Move bDateTime one whole second forward
        bDateTime = new DateTime(2005, 6, 9, 12, 30, 46, 0, DateTimeZone.UTC);

        // bDateTime is later → compare(b,a) must be > 0 (we expect exactly 1)
        assertEquals("Second‑Greater (b > a)", 1,
                     cSecond.compare(bDateTime, aDateTime));

        // aDateTime is earlier → compare(a,b) must be < 0 (we expect exactly -1)
        assertEquals("Second‑Less (a < b)", -1,
                     cSecond.compare(aDateTime, bDateTime));

        // Verify the same ordering with java.util.Date instances
        assertEquals("Second‑Greater (Date) (b > a)", 1,
                     cSecond.compare(bDateTime.toDate(), aDateTime.toDate()));
        assertEquals("Second‑Less (Date) (a < b)", -1,
                     cSecond.compare(aDateTime.toDate(), bDateTime.toDate()));
    }

    // -----------------------------------------------------------------------
    // Helper methods (if needed by other tests – kept here for completeness)
    // -----------------------------------------------------------------------
    private DateTime getADate(String s) {
        // Example helper – not used in this particular test
        return new DateTime(Long.parseLong(s), ISO);
    }

    private java.util.List<String> loadAList(String[] someStrs) {
        java.util.List<String> list = new java.util.ArrayList<String>();
        java.util.Collections.addAll(list, someStrs);
        return list;
    }

    private boolean isListSorted(java.util.List<?> tl) {
        for (int i = 1; i < tl.size(); i++) {
            if (((Comparable) tl.get(i - 1)).compareTo(tl.get(i)) > 0) {
                return false;
            }
        }
        return true;
    }
}
