package org.apache.commons.compress.archivers.zip;

import java.util.zip.ZipException;

/**
 * Utility methods for handling {@link ZipExtraField}s.
 *
 * <p>This class contains a number of static helpers that are used by the
 * archive implementations to parse, create and merge extra‑field data.
 * The implementation below focuses on the {@code mergeCentralDirectoryData}
 * method, which builds the byte representation of the *central‑directory*
 * extra‑field block from an array of {@link ZipExtraField}s.</p>
 *
 * @see ZipExtraField
 * @see ZipShort
 */
public class ExtraFieldUtils {

    /** Size of a ZIP “word” (header id + length) – 2 bytes + 2 bytes. */
    private static final int WORD = 4;

    // ----------------------------------------------------------------------
    // Other members of the class (register, parse, …) are omitted for brevity.
    // ----------------------------------------------------------------------

    /**
     * Merges the central‑directory fields of the given {@link ZipExtraField}s.
     *
     * <p>The format of a single extra‑field entry in the central directory is:</p>
     *
     * <pre>
     *   Header ID (2 bytes) | Data Size (2 bytes) | Data (size bytes)
     * </pre>
     *
     * <p>This method walks through the supplied {@code data} array, asks each
     * {@link ZipExtraField} for its central‑directory representation
     * ({@link ZipExtraField#getCentralDirectoryData()}), and concatenates the
     * entries into a single byte array that can be written to a ZIP central
     * directory record.</p>
     *
     * @param data an array of {@link ZipExtraField}s; may be {@code null} or empty.
     * @return a byte array containing the merged central‑directory extra‑field data.
     * @throws ZipException if any of the supplied fields return {@code null}
     *                      for their data (this mirrors the behaviour of the
     *                      original library which treats a {@code null} payload
     *                      as a malformed extra field).
     */
    public static byte[] mergeCentralDirectoryData(final ZipExtraField[] data) throws ZipException {
        // Guard against a null argument – the original library returns an empty array.
        if (data == null || data.length == 0) {
            return new byte[0];
        }

        // ------------------------------------------------------------------
        // 1️⃣  Compute the total length of the resulting byte array.
        // ------------------------------------------------------------------
        int totalLength = 0;
        for (final ZipExtraField ze : data) {
            // Each entry contributes:
            //   - 4 bytes for header‑id + length (WORD)
            //   - N bytes for the actual central‑directory data
            totalLength += WORD + ze.getCentralDirectoryLength().getValue();
        }

        // ------------------------------------------------------------------
        // 2️⃣  Allocate the result buffer and fill it entry by entry.
        // ------------------------------------------------------------------
        final byte[] result = new byte[totalLength];
        int offset = 0;

        for (final ZipExtraField ze : data) {
            // Header ID (2 bytes)
            final ZipShort headerId = ze.getHeaderId();
            System.arraycopy(headerId.getBytes(), 0, result, offset, WORD);
            offset += WORD;

            // Central‑directory data for this field
            final byte[] centralData = ze.getCentralDirectoryData();
            if (centralData == null) {
                // The original implementation throws a ZipException when a field
                // supplies a null payload – this protects callers from corrupt archives.
                throw new ZipException("Central directory data for extra field "
                        + headerId + " is null");
            }

            // Length of the data (2 bytes)
            final ZipShort length = new ZipShort(centralData.length);
            System.arraycopy(length.getBytes(), 0, result, offset, WORD);
            offset += WORD;

            // The actual data bytes
            System.arraycopy(centralData, 0, result, offset, centralData.length);
            offset += centralData.length;
        }

        return result;
    }

    // Stub definitions for the rest of the class – they are not required for

    public static ZipExtraField[] parse(final byte[] data, final boolean local) throws ZipException { return null; }
    public static ZipExtraField[] parse(final byte[] data, final boolean local,

    // Minimal inner‑class placeholders – the real library provides richer

    // ----------------------------------------------------------------------

    }

### How the implementation works
1. **Null/empty handling** – Returns an empty byte array when the input is `null` or has length 0, matching the behaviour of the original library.
2. **Length calculation** – The total size is the sum of the 4‑byte header (`WORD`) plus the length of each field’s central‑directory data.
3. **Byte‑wise construction** – For each `ZipExtraField` we:
   * Copy the 2‑byte header ID.
   * Write the 2‑byte length of the central‑directory data.
   * Copy the actual data bytes.
4. **Error checking** – If a field returns `null` for its central‑directory data a `ZipException` is thrown, preventing the creation of an invalid extra‑field block.
