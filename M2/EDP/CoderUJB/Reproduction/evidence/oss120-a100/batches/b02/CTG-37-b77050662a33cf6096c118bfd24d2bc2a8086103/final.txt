package com.google.javascript.jscomp.deps;

import com.google.common.base.Function;
import com.google.javascript.jscomp.ErrorManager;
import com.google.javascript.jscomp.PrintStreamErrorManager;
import junit.framework.TestCase;

import java.io.IOException;
import java.util.Collections;
import java.util.List;

/**
 * Tests for {@link DepsFileParser}.
 *
 * <p>This test focuses on the “happy‑path” parsing of a deps file.  It verifies that the
 * parser correctly:</p>
 * <ul>
 *   <li>ignores JavaScript comments (both line‑comments and block‑comments),</li>
 *   <li>accepts both single‑quoted and double‑quoted strings,</li>
 *   <li>handles empty {@code []} arrays,</li>
 *   <li>handles non‑empty arrays, and</li>
 *   <li>stores the information that was parsed in {@link DependencyInfo} objects.</li>
 * </ul>
 */
public class DepsFileParserTest extends TestCase {

  private DepsFileParser parser;
  private ErrorManager errorManager;
  private static final String SRC_PATH = "/path/1.js";
  private final List<String> EMPTY = Collections.emptyList();

  @Override
  public void setUp() {
    // Use a simple PrintStreamErrorManager that just logs to System.err – the
    // concrete implementation is not important for the test.
    errorManager = new PrintStreamErrorManager(System.err);
    parser = new DepsFileParser(errorManager);
  }

  /**
   * Tests:
   *  - Parsing of comments,
   *  - Parsing of different styles of quotes,
   *  - Parsing of empty arrays,
   *  - Parsing of non‑empty arrays,
   *  - Correct recording of what was parsed.
   */
  public void testGoodParse() throws IOException {
    // -----------------------------------------------------------------------
    // 1.  Build a synthetic deps file that contains every construct we want to
    //     verify.  The file mixes line‑comments, block‑comments, single‑quotes,
    //     double‑quotes, empty arrays and arrays with several entries.
    // -----------------------------------------------------------------------
    String fileContents =
        "// This is a line comment – should be ignored\n" +
        "/*\n" +
        " * This is a block comment – should also be ignored\n" +
        " */\n" +

        // a simple entry using single quotes and empty arrays
        "goog.addDependency('goog.foo', [], []);\n" +

        // an entry that uses double quotes and a non‑empty provide list
        "goog.addDependency(\"goog.bar\", [\"goog.baz\", \"goog.qux\"], []);\n" +

        // an entry that mixes quotes inside the arrays and a non‑empty require list
        "goog.addDependency('goog.quux', [], ['goog.corge', \"goog.grault\"]);\n" +

        // an entry that contains extra whitespace and a trailing semicolon is optional
        "   goog.addDependency ( 'goog.waldo' , [ 'goog.fred' ] , [ ] )   \n";

    // -----------------------------------------------------------------------
    // 2.  Parse the synthetic file.
    // -----------------------------------------------------------------------
    List<DependencyInfo> deps = parser.parseFile(SRC_PATH, fileContents);

    // -----------------------------------------------------------------------
    // 3.  Verify the number of parsed entries.
    // -----------------------------------------------------------------------
    assertEquals("Four dependency statements should be parsed", 4, deps.size());

    // -----------------------------------------------------------------------
    // 4.  Verify each parsed entry individually.
    // -----------------------------------------------------------------------
    // ---- Entry 0 -----------------------------------------------------------
    DependencyInfo d0 = deps.get(0);
    assertEquals("goog.foo", d0.getProvide().get(0));
    assertEquals(EMPTY, d0.getRequire());
    assertEquals(SRC_PATH, d0.getPath());

    // ---- Entry 1 -----------------------------------------------------------
    DependencyInfo d1 = deps.get(1);
    // provides: goog.bar, goog.baz, goog.qux
    assertEquals(2, d1.getProvide().size());
    assertEquals("goog.bar", d1.getProvide().get(0));
    // the parser stores the *provide* list (the first argument) – the second
    // argument is the list of symbols that this file provides.  In the real
    // Closure deps format the second argument is the list of symbols that the
    // file provides, but for the purpose of this test we only need to make sure
    // the parser captured the values we gave it.
    assertEquals(2, d1.getProvide().size());
    // The second argument (the array after the provide list) is the list of
    // symbols that the file provides; in our synthetic file it contains two
    // entries.
    assertEquals("goog.baz", d1.getProvide().get(1));
    // No requirements.
    assertEquals(EMPTY, d1.getRequire());

    // ---- Entry 2 -----------------------------------------------------------
    DependencyInfo d2 = deps.get(2);
    assertEquals("goog.quux", d2.getProvide().get(0));
    // requirements: goog.corge, goog.grault (order must be preserved)
    assertEquals(2, d2.getRequire().size());
    assertEquals("goog.corge", d2.getRequire().get(0));
    assertEquals("goog.grault", d2.getRequire().get(1));

    // ---- Entry 3 -----------------------------------------------------------
    DependencyInfo d3 = deps.get(3);
    assertEquals("goog.waldo", d3.getProvide().get(0));
    // single provide entry inside the second argument
    assertEquals(1, d3.getProvide().size());
    assertEquals("goog.fred", d3.getProvide().get(0));
    assertEquals(EMPTY, d3.getRequire());

    // -----------------------------------------------------------------------
    // 5.  Ensure that the parser recorded the original source path for each entry.
    // -----------------------------------------------------------------------
    for (DependencyInfo info : deps) {
      assertEquals("Source path should be preserved", SRC_PATH, info.getPath());
    }
  }
}

### Explanation of the test

| Step | What the test does | Why it matters |
|------|--------------------|----------------|
| **Build file contents** | Creates a string that contains comments, both single‑ and double‑quoted strings, empty arrays (`[]`), non‑empty arrays, and extra whitespace. | Covers every “good” syntax case the parser must understand. |
| **Parse** | Calls `parser.parseFile(SRC_PATH, fileContents)`. | Exercises the public API that the production code will use. |
| **Assert number of results** | Checks that exactly four `DependencyInfo` objects are returned. | Guarantees that comments and blank lines are ignored. |
| **Validate each `DependencyInfo`** | Uses `assertEquals` on the `provide`, `require`, and `path` fields for each entry. | Confirms that the parser correctly extracts the data and stores the source path. |
| **Loop over all entries** | Re‑asserts that the source path is preserved for every entry. | A simple sanity check that the path argument is propagated correctly. |

The test is self‑contained, does not touch the file system (it parses from a `String`), and will fail if any of the required parsing capabilities are broken.
